// Package cryptosuite adapts implemented SM2/SM3/SM4 and ML-KEM components;
// unsupported suites remain fail-closed. Implementation is not certification.
package cryptosuite

import (
        "crypto/cipher"
        "errors"
        "hash"
)

var ErrNotImplemented = errors.New("secure cryptographic backend/handshake is not implemented; refusing operation")

type Role uint8

const (
        Initiator Role = 1
        Responder Role = 2
)

// A handle points to private state owned by an adapter. Never log it.
type PrivateHandle interface{ Destroy() error }
type IdentityPublic struct {
        ID, SignKeyID, AgreementKeyID          string
        SM2UserID, SignPublic, AgreementPublic []byte
}
type Ephemeral struct {
        Public  []byte
        Private PrivateHandle
}
type ExchangeResult struct {
        SharedSecret []byte // exactly 32 bytes for this proposed suite
        // Exactly 32 bytes each: initiator sends S2 and expects S1; responder
        // sends S1 and expects S2. Required by handshake body version 2.
        SendConfirmation, ExpectedPeerConfirmation []byte
}

// Implement the full SM2 key exchange, not merely ordinary ECDH on an SM2 curve.
// The caller must verify ExpectedPeerConfirmation before admitting a Channel.
type SM2KeyAgreement interface {
        GenerateEphemeral() (Ephemeral, error)
        Derive(role Role, local IdentityPublic, localStatic PrivateHandle, localEphemeral Ephemeral, peer IdentityPublic, peerEphemeral []byte) (ExchangeResult, error)
}
type SM2Identity interface {
        Generate() (signPublic []byte, private PrivateHandle, err error)
        Sign(private PrivateHandle, userID, message []byte) (signature []byte, err error)
        Verify(public, userID, message, signature []byte) error
}
type KEM interface {
        Generate() (public []byte, private PrivateHandle, err error)
        Encapsulate(public []byte) (sharedSecret, ciphertext []byte, err error)
        Decapsulate(private PrivateHandle, ciphertext []byte) (sharedSecret []byte, err error)
}
type SMPrimitives interface {
        NewSM3() hash.Hash
        NewSM4GCM(key []byte) (cipher.AEAD, error)
}

// Suite identifiers. They mirror policy.Suite values and are repeated here so
// the adapter layer can answer "is a real backend wired for this suite?" without
// importing the policy package.
const (
        SuiteTraditional uint16 = 1
        SuiteHybrid      uint16 = 2
)

// suiteReadiness records, per suite, whether a REAL cryptographic backend is
// wired. It is a statement about the adapter layer only — a true entry here does
// not by itself mean a handshake exists or is verified.
//
// Suite 1 (traditional SM2 + HKDF-SM3 + SM4-GCM): adapters implemented and
// verified at the component level.
//
// Suite 2 (hybrid, SM2 + ML-KEM-768): NOW TRUE, opened in M3.
//
// This was deliberately left false through M2, M2.1 and the whole of M3's
// implementation, and was only flipped once EVERY acceptance condition had
// actually passed. It was never flipped to make a failing test go green:
//
//  1. a real ML-KEM-768 exchange reaches the handshake   - hybrid_test.go
//  2. Suite 2 end-to-end succeeds                        - scripts/m2-e2e.sh MODE=hybrid
//  3. the result depends on BOTH Zsm2 and Zpq            - keyschedule_hybrid_test.go A/B
//  4. implicit rejection is caught by Finished           - kem_implicit_test.go
//  5. wire-level downgrade is rejected                   - downgrade_test.go (9 cases)
//  6. no automatic fallback to suite 1                   - no_fallback_test.go
//  7. Suite 1 regression still passes                    - full suite + MODE=traditional
//
// WHAT TRUE HERE DOES AND DOES NOT MEAN: it means the hybrid handshake exists and
// consumes a real KEM shared secret. It does NOT mean the system is
// post-quantum-secure — SM2 IDENTITY AUTHENTICATION is not post-quantum — and it
// does not mean the hybrid construction has been audited or proved.
var suiteReadiness = map[uint16]bool{
        SuiteTraditional: true,
        SuiteHybrid:      true,
}

// RequireImplemented reports whether a real backend is wired for suite.
//
// This REPLACES the original coarse placeholder `RequireImplemented() error`,
// which returned an error unconditionally and told callers nothing about which
// capability was missing. The replacement is deliberately not a global
// readiness flag that can be flipped to nil: it answers per suite, and the
// suites it accepts are exactly the ones whose adapters exist and whose
// integration tests pass. An unknown suite is refused.
func RequireImplemented(suite uint16) error {
        if ok, known := suiteReadiness[suite]; known && ok {
                return nil
        }
        return ErrNotImplemented
}

// ImplementedSuites lists the suites with a real adapter backend, in preference
// order: the hybrid suite first, because a peer that supports both must prefer
// it.
func ImplementedSuites() []uint16 {
        return []uint16{SuiteHybrid, SuiteTraditional}
}

// mlkem768 implements KEM over the Go standard library's crypto/mlkem package
// (FIPS 203 ML-KEM-768). The standard library is used rather than a third-party
// implementation so that the post-quantum half of the hybrid suite rests on the
// Go toolchain's own audited code.
//
// Two properties that callers must not get wrong are handled explicitly here:
//
//  1. Encapsulate returns (sharedSecret, ciphertext) in that order. Swapping
//     them compiles silently and produces a broken protocol, so the adapter
//     encodes the order once, in one place.
//
//  2. FIPS 203 defines implicit rejection: decapsulating a ciphertext that has
//     the correct length but is not a genuine encryption of the key returns a
//     DIFFERENT pseudorandom shared secret with err == nil. There is no error to
//     check. A protocol that treats "no error" as "authenticated" would accept
//     forged ciphertexts. This function therefore returns exactly what the
//     library returns and never invents an error; the handshake must detect the
//     mismatch later, through the Finished/confirmation values.
type mlkem768 struct{}

// NewMLKEM768 returns the ML-KEM-768 KEM adapter.
func NewMLKEM768() KEM { return mlkem768{} }

// Generate creates a fresh ML-KEM-768 key pair.
//
// For a handshake, callers MUST generate a new ephemeral key per handshake;
// reusing a decapsulation key across handshakes is a protocol error, not just
// hygiene.
func (mlkem768) Generate() ([]byte, PrivateHandle, error) {
        dk, err := mlkem.GenerateKey768()
        if err != nil {
                return nil, nil, err
        }
        pub := dk.EncapsulationKey().Bytes()
        if len(pub) != MLKEM768EncapKeySize {
                return nil, nil, ErrInvalidLength
        }
        return pub, &mlkemPrivateKey{dk: dk}, nil
}

// Encapsulate produces (sharedSecret, ciphertext) for the peer's public key.
func (mlkem768) Encapsulate(public []byte) ([]byte, []byte, error) {
        if len(public) != MLKEM768EncapKeySize {
                return nil, nil, ErrInvalidLength
        }
        ek, err := mlkem.NewEncapsulationKey768(public)
        if err != nil {
                return nil, nil, ErrInvalidKey
        }
        shared, ct := ek.Encapsulate()
        if len(shared) != MLKEM768SharedSize || len(ct) != MLKEM768CiphertextSize {
                return nil, nil, ErrInvalidLength
        }
        return shared, ct, nil
}

// Decapsulate recovers the shared secret from a ciphertext.
//
// IMPORTANT: a well-formed-length ciphertext that was tampered with will NOT
// produce an error. It produces a different shared secret (FIPS 203 implicit
// rejection). The returned error only signals a structurally unusable input,
// such as a ciphertext of the wrong length. Callers must not use err == nil as
// evidence that the KEM exchange succeeded; only the authenticated handshake
// transcript can establish that.
func (mlkem768) Decapsulate(handle PrivateHandle, ciphertext []byte) ([]byte, error) {
        k, ok := handle.(*mlkemPrivateKey)
        if !ok || k == nil || k.dk == nil {
                return nil, ErrInvalidKey
        }
        if len(ciphertext) != MLKEM768CiphertextSize {
                return nil, ErrInvalidLength
        }
        shared, err := k.dk.Decapsulate(ciphertext)
        if err != nil {
                return nil, err
        }
        if len(shared) != MLKEM768SharedSize {
                return nil, ErrInvalidLength
        }
        return shared, nil
}

// mlkemPrivateKey wraps the decapsulation key. The standard library does not
// expose a wipe API, so Destroy only drops the reachable reference.
type mlkemPrivateKey struct {
        dk *mlkem.DecapsulationKey768
}

func (k *mlkemPrivateKey) Destroy() error {
        if k == nil {
                return nil
        }
        k.dk = nil
        return nil
}

// Public returns the encapsulation key for this decapsulation key.
func (k *mlkemPrivateKey) Public() ([]byte, error) {
        if k == nil || k.dk == nil {
                return nil, ErrKeyDestroyed
        }
        return k.dk.EncapsulationKey().Bytes(), nil
}

// This file adds the minimum surface a trusted local keystore needs: import a
// previously generated scalar, and export the scalar of a handle we generated.
//
// It deliberately does NOT widen the generic PrivateHandle interface. Handshake
// code receives a PrivateHandle and can only ask it to sign, to agree, or to be
// destroyed. Being able to pull a raw scalar out of a handle is a capability
// that only provisioning code should hold, so it is a separate, explicitly
// named interface that callers must type-assert to.
//
// The exported scalar is secret material. Callers must write it only into a
// trusted, access-controlled key file and must never log it, print it, pass it
// on a command line, or place it in results/CSV/Git.

// KeyKind identifies the trust role a private handle serves. A signing key and
// a static agreement key have the same byte-encoding shape, so the encoding
// cannot tell them apart; the handle itself must.
type KeyKind uint8

const (
        KeyKindSigning   KeyKind = 1
        KeyKindAgreement KeyKind = 2
)

// KindedHandle is implemented by handles that know which trust role they serve.
type KindedHandle interface {
        PrivateHandle
        Kind() KeyKind
}

func (k *sm2PrivateKey) Kind() KeyKind  { return KeyKindSigning }
func (s *sm2StaticKey) Kind() KeyKind   { return KeyKindAgreement }
func (e *ecdhPrivateKey) Kind() KeyKind { return KeyKindAgreement }

// PublicHandle is implemented by private handles that can report the public
// value corresponding to their own private scalar. It lets a caller obtain its
// own public key from the handle it already holds, instead of reading it back
// out of a configuration file where it could disagree.
type PublicHandle interface {
        PrivateHandle
        Public() ([]byte, error)
}

func (k *sm2PrivateKey) Public() ([]byte, error) {
        pub, err := k.public()
        if err != nil {
                return nil, err
        }
        return encodePublicKey(pub)
}

func (e *ecdhPrivateKey) Public() ([]byte, error) {
        k, err := e.ecdhKey()
        if err != nil {
                return nil, err
        }
        return k.PublicKey().Bytes(), nil
}

// ScalarHandle is implemented by private handles whose raw scalar can be
// exported for trusted local storage.
type ScalarHandle interface {
        PrivateHandle
        // Scalar returns a fresh copy of the raw private scalar (32 bytes for every
        // SM2/SM2-exchange key in this suite). Treat the result as secret.
        Scalar() ([]byte, error)
}

func (k *sm2PrivateKey) Scalar() ([]byte, error) {
        k.mu.Lock()
        defer k.mu.Unlock()
        if k.priv == nil || k.priv.D == nil {
                return nil, ErrKeyDestroyed
        }
        out := make([]byte, SM2PrivateSize)
        k.priv.D.FillBytes(out)
        return out, nil
}

func (s *sm2StaticKey) Scalar() ([]byte, error) {
        k, err := s.ecdhKey()
        if err != nil {
                return nil, err
        }
        b := k.Bytes()
        if len(b) != SM2PrivateSize {
                return nil, ErrInvalidLength
        }
        return append([]byte(nil), b...), nil
}

func (e *ecdhPrivateKey) Scalar() ([]byte, error) {
        k, err := e.ecdhKey()
        if err != nil {
                return nil, err
        }
        b := k.Bytes()
        if len(b) != SM2PrivateSize {
                return nil, ErrInvalidLength
        }
        return append([]byte(nil), b...), nil
}

// NewSM2IdentityFromScalar imports an SM2 signing key from a 32-byte scalar and
// returns its uncompressed public encoding alongside a fresh handle.
//
// The scalar must be in [1, n-2] per GB/T 32918.1-2016; the upstream library
// rejects out-of-range values rather than silently reducing them.
func NewSM2IdentityFromScalar(scalar []byte) ([]byte, PrivateHandle, error) {
        if len(scalar) != SM2PrivateSize {
                return nil, nil, ErrInvalidLength
        }
        priv, err := sm2.NewPrivateKey(scalar)
        if err != nil {
                return nil, nil, ErrInvalidKey
        }
        // Guard against a scalar that the upstream accepted but that is not actually
        // a usable SM2 key, so the keystore never records a key that cannot sign.
        if !sm2.IsSM2PublicKey(&priv.PublicKey) {
                return nil, nil, ErrInvalidKey
        }
        pub, err := encodePublicKey(&priv.PublicKey)
        if err != nil {
                return nil, nil, err
        }
        return pub, &sm2PrivateKey{priv: priv}, nil
}

// GenerateSM2IdentityKeyPair generates a fresh signing key pair and returns both
// the public encoding and the raw scalar, so a trusted initializer can persist
// the scalar in one step without an extra export round trip.
func GenerateSM2IdentityKeyPair() (public, scalar []byte, err error) {
        pub, handle, err := NewSM2Identity().Generate()
        if err != nil {
                return nil, nil, err
        }
        s, ok := handle.(ScalarHandle)
        if !ok {
                _ = handle.Destroy()
                return nil, nil, ErrInvalidKey
        }
        scalar, err = s.Scalar()
        if err != nil {
                _ = handle.Destroy()
                return nil, nil, err
        }
        if err = handle.Destroy(); err != nil {
                return nil, nil, err
        }
        return pub, scalar, nil
}

// GenerateSM2AgreementKeyPair generates a fresh static agreement key pair and
// returns both the public encoding and the raw scalar. This is a DIFFERENT key
// type from the signing key and must never be interchanged with it.
func GenerateSM2AgreementKeyPair() (public, scalar []byte, err error) {
        pub, handle, err := NewSM2StaticKey()
        if err != nil {
                return nil, nil, err
        }
        s, ok := handle.(ScalarHandle)
        if !ok {
                _ = handle.Destroy()
                return nil, nil, ErrInvalidKey
        }
        scalar, err = s.Scalar()
        if err != nil {
                _ = handle.Destroy()
                return nil, nil, err
        }
        if err = handle.Destroy(); err != nil {
                return nil, nil, err
        }
        return pub, scalar, nil
}

// sm2Agreement implements SM2KeyAgreement using the full GB/T 32918.3 SM2 key
// exchange as provided by gmsm's ecdh package (SM2MQV + SM2SharedKey), with
// standard S1/S2 hash encoding composed using upstream SM3 and SM2ZA.
//
// This is deliberately NOT ordinary ECDH on the SM2 curve. Plain ECDH would
// compute a single scalar multiplication and skip:
//   - the implicit signature / MQV-style combination of static AND ephemeral
//     keys on each side,
//   - the SM2 confirmation value derived from w = avf(x) truncation,
//   - the ZA/ZB ordering, which is what ties the shared key to both identities
//     and to the initiator/responder roles.
//
// Substituting ECDH would therefore produce a value that "looks like" a shared
// secret while providing none of those bindings. AGENTS.md forbids that
// substitution, so this adapter only ever calls SM2MQV/SM2SharedKey.
//
// Role and ordering: SM2SharedKey's isResponder flag selects whether ZA||ZB or
// ZB||ZA is fed into the KDF. That flag must match the peer's role, so a role
// swap between the two sides changes the derived key and is detectable.
type sm2Agreement struct{}

// NewSM2Agreement returns the SM2 key-agreement adapter.
func NewSM2Agreement() SM2KeyAgreement { return sm2Agreement{} }

// GenerateEphemeral creates a fresh SM2 ephemeral key pair for one handshake.
// Ephemeral keys are never reused across handshakes.
func (sm2Agreement) GenerateEphemeral() (Ephemeral, error) {
        k, err := ecdh.P256().GenerateKey(rand.Reader)
        if err != nil {
                return Ephemeral{}, err
        }
        return Ephemeral{Public: k.PublicKey().Bytes(), Private: &ecdhPrivateKey{key: k}}, nil
}

// Derive runs the full SM2 key exchange and returns the 32-byte shared result.
//
// Parameters are exactly the cryptographic inputs the exchange needs; nothing
// is inferred from local policy, because a policy guess must never influence a
// key derivation.
func (sm2Agreement) Derive(
        role Role,
        local IdentityPublic,
        localStatic PrivateHandle,
        localEphemeral Ephemeral,
        peer IdentityPublic,
        peerEphemeral []byte,
) (ExchangeResult, error) {
        var out ExchangeResult

        if role != Initiator && role != Responder {
                return out, ErrInvalidKey
        }
        if !validUserID(local.SM2UserID) || !validUserID(peer.SM2UserID) {
                return out, ErrInvalidIdentity
        }

        // Static agreement key is a separate handle from the signing key, as
        // required: a compromise of one must not silently compromise the other.
        static, ok := localStatic.(*sm2StaticKey)
        if !ok || static == nil {
                return out, ErrInvalidKey
        }
        sLocal, err := static.ecdhKey()
        if err != nil {
                return out, err
        }

        eph, ok := localEphemeral.Private.(*ecdhPrivateKey)
        if !ok || eph == nil {
                return out, ErrInvalidKey
        }
        eLocal, err := eph.ecdhKey()
        if err != nil {
                return out, err
        }

        // Bind the advertised local ephemeral to its private handle. Confirmation
        // must cover the same RA/RB that the scalar multiplication actually used.
        if !ConstantTimeEqual(localEphemeral.Public, eLocal.PublicKey().Bytes()) {
                return out, ErrInvalidKey
        }
        if len(local.AgreementPublic) != 0 && !ConstantTimeEqual(local.AgreementPublic, sLocal.PublicKey().Bytes()) {
                return out, ErrInvalidKey
        }

        // Peer public values are parsed strictly: wrong length, wrong prefix or an
        // off-curve point is rejected before any scalar multiplication.
        sRemote, err := parseECDHPublicKey(peer.AgreementPublic)
        if err != nil {
                return out, err
        }
        eRemote, err := parseECDHPublicKey(peerEphemeral)
        if err != nil {
                return out, err
        }

        // Reject a peer that echoed our own ephemeral public key back as its own.
        // Without this check, a reflection attack can make both sides agree on a
        // value derived from a single party's ephemeral secret, removing the
        // peer's contribution to freshness.
        if ConstantTimeEqual(localEphemeral.Public, peerEphemeral) {
                return out, ErrInvalidKey
        }

        // uv = [t](sRemote + [x1]eRemote); t = (sLocal + [x2]eLocal) mod n
        uv, err := sLocal.SM2MQV(eLocal, sRemote, eRemote)
        if err != nil {
                return out, ErrInvalidKey
        }

        // The initiator (device) computes ZA||ZB ordering; the responder
        // (gateway) computes ZB||ZA. isResponder must be true on the responder.
        isResponder := role == Responder
        // kenLen is fixed at 32 so the result length is not attacker-influenced.
        secret, err := uv.SM2SharedKey(
                isResponder,
                SM3Size,
                sLocal.PublicKey(),
                sRemote,
                local.SM2UserID,
                peer.SM2UserID,
        )
        if err != nil {
                return out, ErrInvalidKey
        }
        if len(secret) != SM3Size {
                return out, ErrInvalidLength
        }
        out.SharedSecret = secret
        out.SendConfirmation, out.ExpectedPeerConfirmation, err = sm2Confirmations(
                role, uv, sLocal.PublicKey(), sRemote, eLocal.PublicKey(), eRemote,
                local.SM2UserID, peer.SM2UserID,
        )
        if err != nil {
                clear(secret)
                return ExchangeResult{}, err
        }
        return out, nil
}

// sm2Confirmations encodes the GB/T 32918.3 S1/S2 inputs. EC operations, ZA
// and SM3 remain upstream library operations. Coordinates are fixed 32-byte
// big-endian values on the production sm2p256v1 curve; point prefixes are not
// hashed. Canonical A is initiator and B is responder, independent of locality.
func sm2Confirmations(role Role, uv, localS, peerS, localE, peerE *ecdh.PublicKey, localUID, peerUID []byte) ([]byte, []byte, error) {
        za, err := localS.SM2ZA(NewSM3(), localUID)
        if err != nil {
                return nil, nil, ErrInvalidIdentity
        }
        zb, err := peerS.SM2ZA(NewSM3(), peerUID)
        if err != nil {
                return nil, nil, ErrInvalidIdentity
        }
        ra, rb := localE.Bytes(), peerE.Bytes()
        if role == Responder {
                za, zb, ra, rb = zb, za, rb, ra
        }
        u := uv.Bytes()
        defer clear(u)
        h := NewSM3()
        h.Write(u[1:33])
        h.Write(za)
        h.Write(zb)
        h.Write(ra[1:])
        h.Write(rb[1:])
        inner := h.Sum(nil)
        defer clear(inner)
        tag := func(prefix byte) []byte {
                h := NewSM3()
                h.Write([]byte{prefix})
                h.Write(u[33:])
                h.Write(inner)
                return h.Sum(nil)
        }
        s1, s2 := tag(0x02), tag(0x03)
        if role == Responder {
                return s1, s2, nil
        }
        return s2, s1, nil
}

// SignKeyToAgreementPublic derives the SM2 static-agreement public key that
// corresponds to an SM2 SIGNING key. It exists for diagnostics and test
// fixtures only: the protocol requires the two key types to be independently
// generated, and production code must never reuse a signing key as an agreement

func SignKeyToAgreementPublic(signPriv PrivateHandle) ([]byte, error) {

        if !ok {
                return nil, ErrInvalidKey
        }
        pub, err := k.public()
        if err != nil {
                return nil, err
        }
        return encodePublicKey(pub)
}

// parseECDHPublicKey accepts only the canonical uncompressed encoding.
func parseECDHPublicKey(b []byte) (*ecdh.PublicKey, error) {
        if len(b) != SM2CoordinateSize || b[0] != 0x04 {
                return nil, ErrInvalidLength
        }
        k, err := ecdh.P256().NewPublicKey(b)
        if err != nil {
                return nil, ErrInvalidKey
        }
        return k, nil
}

                return nil, ErrInvalidLength
        }
        k, err := ecdh.P256().NewPrivateKey(b)
        if err != nil {
                return nil, ErrInvalidKey
        }
        return k, nil
}

// sm2Identity implements SM2Identity using gmsm's SM2 signature.
//
// Two details are enforced here rather than left to callers, because getting
// either wrong silently changes the security meaning of a handshake:
//
//  1. Signatures are exchanged as fixed-width r||s (64 bytes), NOT ASN.1 DER.
//     DER is variable length, which would break the protocol's strict
//     fixed-field frame parsing and let a peer smuggle length ambiguity into
//     the transcript.
//
//  2. The SM2 user identifier (ZA binding) is mandatory and validated. The ZA
//     value is what binds a signature to an identity; if uid were empty or
//     ignored, two different identities could verify the same signature.
//
// The signing call path is upstream gmsm only. This file performs no curve
// arithmetic.
type sm2Identity struct{}

// NewSM2Identity returns the SM2 identity-signature adapter.
func NewSM2Identity() SM2Identity { return sm2Identity{} }

// Generate produces a fresh SM2 key pair. The returned public value is the
// uncompressed point encoding (0x04 || X || Y, 65 bytes).
func (sm2Identity) Generate() ([]byte, PrivateHandle, error) {
        priv, err := sm2.GenerateKey(rand.Reader)
        if err != nil {
                return nil, nil, err
        }
        pub, err := encodePublicKey(&priv.PublicKey)
        if err != nil {
                return nil, nil, err
        }
        return pub, &sm2PrivateKey{priv: priv}, nil
}

// Sign produces a fixed-width r||s SM2 signature over message, bound to userID.
//
// Signature semantics: priv.Sign(rand, message, NewSM2SignerOption(true, userID))
// passes the RAW message together with uid, and the "forceGMSign" option makes
// the library compute ZA = H(ENTLA || IDA || a || b || xG || yG || xA || yA)
// and hash ZA||M itself. Passing an already-hashed digest here would double
// hash the message, so we always hand over the raw message.
func (sm2Identity) Sign(handle PrivateHandle, userID, message []byte) ([]byte, error) {
        if !validUserID(userID) {
                return nil, ErrInvalidIdentity
        }
        k, ok := handle.(*sm2PrivateKey)
        if !ok || k == nil {
                return nil, ErrInvalidKey
        }
        k.mu.Lock()
        defer k.mu.Unlock()
        if k.priv == nil {
                return nil, ErrKeyDestroyed
        }
        der, err := k.priv.Sign(rand.Reader, message, sm2.NewSM2SignerOption(true, userID))
        if err != nil {
                return nil, err
        }
        r, s, err := splitASN1Signature(der)
        if err != nil {
                return nil, err
        }
        out := make([]byte, 0, SM2SignatureSize)
        out = append(out, leftPad(r, SM2PrivateSize)...)
        out = append(out, leftPad(s, SM2PrivateSize)...)
        return out, nil
}

// Verify checks a fixed-width r||s signature against the peer's public key and
// the declared user identifier, using the upstream verification API.
func (sm2Identity) Verify(public, userID, message, signature []byte) error {
        if !validUserID(userID) {
                return ErrInvalidIdentity
        }
        if len(signature) != SM2SignatureSize {
                return ErrInvalidLength
        }
        pub, err := parseSM2PublicKey(public)
        if err != nil {
                return err
        }
        // VerifyASN1WithSM2 takes the raw message plus uid, mirroring the sign side,
        // so ZA is recalculated from the presented public key. A wrong uid or a
        // substituted public key therefore fails verification rather than being
        // silently tolerated.
        der := joinASN1Signature(signature[:SM2PrivateSize], signature[SM2PrivateSize:])
        if !sm2.VerifyASN1WithSM2(pub, userID, message, der) {
                return ErrVerificationFail
        }
        return nil
}

// encodePublicKey renders an SM2 public key as the uncompressed point encoding.
func encodePublicKey(pub *ecdsa.PublicKey) ([]byte, error) {
        if pub == nil || pub.X == nil || pub.Y == nil {
                return nil, ErrInvalidKey
        }
        out := make([]byte, 0, SM2CoordinateSize)
        out = append(out, 0x04)
        out = append(out, leftPad(pub.X.Bytes(), SM2PrivateSize)...)
        out = append(out, leftPad(pub.Y.Bytes(), SM2PrivateSize)...)
        if len(out) != SM2CoordinateSize {
                return nil, ErrInvalidLength
        }
        return out, nil
}

// parseSM2PublicKey validates an uncompressed point encoding strictly: exact
// length, correct prefix, and an on-curve point. An off-curve or malformed
// public key must be rejected before it reaches any protocol logic.
func parseSM2PublicKey(public []byte) (*ecdsa.PublicKey, error) {
        if len(public) != SM2CoordinateSize || public[0] != 0x04 {
                return nil, ErrInvalidLength
        }
        pub, err := sm2.ParseUncompressedPublicKey(public)
        if err != nil {
                return nil, ErrInvalidKey
        }
        if !sm2.IsSM2PublicKey(pub) {
                return nil, ErrInvalidKey
        }
        return pub, nil
}

// sm2PrivateKey wraps the upstream key so Destroy has a defined, documented
// meaning. Go managed memory cannot guarantee physical erasure (library copies,
// stack copies, GC copies, swap and crash dumps all remain possible); this only
// ends use of the handle and drops the reachable scalar.
type sm2PrivateKey struct {
        mu   sync.Mutex
        priv *sm2.PrivateKey
}

func (k *sm2PrivateKey) Destroy() error {
        k.mu.Lock()
        defer k.mu.Unlock()
        if k.priv == nil {
                return nil
        }
        if k.priv.D != nil {
                k.priv.D.SetInt64(0)
        }
        k.priv = nil
        return nil
}

// public returns the wrapped public key. It returns a shallow copy of just the
// ecdsa.PublicKey field: sm2.PrivateKey embeds sync.Once, so copying the whole
// struct would duplicate a lock (flagged by go vet).
func (k *sm2PrivateKey) public() (*ecdsa.PublicKey, error) {
        k.mu.Lock()
        defer k.mu.Unlock()
        if k.priv == nil {
                return nil, ErrKeyDestroyed
        }
        cp := k.priv.PublicKey
        return &cp, nil
}

func (k *sm2PrivateKey) value() (*sm2.PrivateKey, error) {
        k.mu.Lock()
        defer k.mu.Unlock()
        if k.priv == nil {
                return nil, ErrKeyDestroyed
        }
        return k.priv, nil
}

// --- ASN.1 <-> fixed-width signature conversion -----------------------------

type asn1Signature struct{ R, S *big.Int }

func splitASN1Signature(der []byte) (r, s []byte, err error) {
        var v asn1Signature
        if _, err = asn1.Unmarshal(der, &v); err != nil {
                return nil, nil, ErrInvalidLength
        }
        if v.R == nil || v.S == nil {
                return nil, nil, ErrInvalidLength
        }
        if v.R.Sign() <= 0 || v.S.Sign() <= 0 {
                return nil, nil, ErrInvalidLength
        }
        return v.R.Bytes(), v.S.Bytes(), nil
}

func joinASN1Signature(r, s []byte) []byte {
        v := asn1Signature{R: new(big.Int).SetBytes(r), S: new(big.Int).SetBytes(s)}
        der, err := asn1.Marshal(v)
        if err != nil {
                return nil
        }
        return der
}

func leftPad(b []byte, size int) []byte {
        if len(b) >= size {
                return b[len(b)-size:]
        }
        out := make([]byte, size)
        copy(out[size-len(b):], b)
        return out
}

// NewSM3 returns a fresh SM3 hash. GB/T 32905-2016.
func NewSM3() hash.Hash { return sm3.New() }

// SumSM3 is a convenience one-shot SM3 over data.
func SumSM3(data []byte) []byte {
        s := sm3.Sum(data)
        return s[:]
}

// HMACSM3 computes HMAC using SM3 as the underlying hash, per RFC 2104 applied
// to SM3. The gmsm library provides no HMAC-SM3 wrapper, so the standard
// construction is expressed with crypto/hmac over the upstream SM3 hash. This is
// composition of a vetted hash with a vetted HMAC construction, not a new
// primitive.
func HMACSM3(key, data []byte) []byte {
        m := hmac.New(sm3.New, key)
        m.Write(data)
        return m.Sum(nil)
}

// SM3KDF is the GB/T 32918.3-2016 KDF (counter-mode SM3). It is exposed because
// the SM2 key-exchange result uses it, and it is NOT the same construction as
// HKDF. Do not use it as a general-purpose HKDF substitute.
func SM3KDF(z []byte, keyLen int) []byte {
        if keyLen <= 0 {
                return nil
        }
        return sm3.Kdf(z, keyLen)
}

// hkdfExtract is HKDF-Extract instantiated with HMAC-SM3, following RFC 5869
// section 2.2 exactly (including the all-zero-salt rule).
func hkdfExtract(salt, ikm []byte) []byte {
        if len(salt) == 0 {
                salt = make([]byte, sm3.Size)
        }
        return HMACSM3(salt, ikm)
}

// hkdfExpand is HKDF-Expand instantiated with HMAC-SM3, following RFC 5869
// section 2.3. The loop bound of 255 blocks is enforced rather than assumed.
func hkdfExpand(prk, info []byte, outLen int) ([]byte, error) {
        if outLen <= 0 || outLen > 255*sm3.Size {
                return nil, ErrInvalidLength
        }
        out := make([]byte, 0, outLen)
        var prev []byte
        var counter [1]byte
        for n := byte(1); len(out) < outLen; n++ {
                m := hmac.New(sm3.New, prk)
                m.Write(prev)
                m.Write(info)
                counter[0] = n
                m.Write(counter[:])
                prev = m.Sum(nil)
                out = append(out, prev...)
        }
        return out[:outLen], nil
}

// HKDFExtractSM3 exposes HKDF-Extract for tests and for callers that need the
// split extract/expand steps.
func HKDFExtractSM3(salt, ikm []byte) []byte { return hkdfExtract(salt, ikm) }

// HKDFExpandSM3 exposes HKDF-Expand over an existing PRK.
func HKDFExpandSM3(prk, info []byte, outLen int) ([]byte, error) {
        return hkdfExpand(prk, info, outLen)
}

// HKDFSM3 runs the full extract-then-expand schedule.
func HKDFSM3(salt, ikm, info []byte, outLen int) ([]byte, error) {
        return hkdfExpand(hkdfExtract(salt, ikm), info, outLen)
}

// Encode implements E(tag, fields...) from the protocol draft: a domain-separated
// encoding with an explicit per-field type tag and a 4-byte big-endian length.
// Every field is length-prefixed, so no concatenation is ambiguous and no
// unbounded string join can collide. This function is what makes identity,
// role, suite and transcript binding unambiguous.
//
// Wire form: len32(tag) || tag || len32 || field || len32 || field ...
func Encode(tag string, fields ...[]byte) []byte {
        out := make([]byte, 0, 16+len(tag)+8*len(fields))
        out = appendLen(out, []byte(tag))
        for _, f := range fields {
                out = appendLen(out, f)
        }
        return out
}

// EncodeString is Encode for a single string field.
func EncodeString(tag, value string) []byte { return Encode(tag, []byte(value)) }

// EncodeSuite binds a suite identifier as a 2-byte big-endian field so that a
// suite change provably changes every derived output.
func EncodeSuite(tag string, suite uint16) []byte {
        var b [2]byte
        binary.BigEndian.PutUint16(b[:], suite)
        return Encode(tag, b[:])
}

func appendLen(dst, field []byte) []byte {
        var n [4]byte
        binary.BigEndian.PutUint32(n[:], uint32(len(field)))
        dst = append(dst, n[:]...)
        return append(dst, field...)
}

// ConstantTimeEqual compares two byte slices without early exit. Callers MUST
// use this (or an upstream verification API) for tags, confirmations and any
// other secret-dependent comparison; bytes.Equal is not sufficient.
func ConstantTimeEqual(a, b []byte) bool {
        return subtle.ConstantTimeCompare(a, b) == 1
}

// sm4Primitives implements SMPrimitives over the upstream SM4 block cipher.
//
// SM4-GCM here is SM4 in Galois/Counter Mode. It is NOT AES-GCM with a
// different key size: the block cipher is SM4 (GB/T 32907-2016) and only the
// GCM mode construction is shared. Substituting AES would change every
// ciphertext and every tag, and is forbidden by AGENTS.md.
//
// The size limits below are what the record layer relies on. crypto/cipher's
// NewGCM standardises nonce to 12 bytes and tag to 16 bytes, and this adapter
// verifies that at construction time rather than trusting it silently.
type sm4Primitives struct{}

// NewSMPrimitives returns the SM3/SM4 primitive factory.
func NewSMPrimitives() SMPrimitives { return sm4Primitives{} }

func (sm4Primitives) NewSM3() hash.Hash { return NewSM3() }

// NewSM4GCM builds an AEAD over an exact 16-byte SM4 key.
func (sm4Primitives) NewSM4GCM(key []byte) (cipher.AEAD, error) {
        return NewSM4GCM(key)
}

// NewSM4GCM constructs SM4-GCM with the protocol's fixed parameters and
// validates them, so a future library change cannot silently alter nonce or tag
// sizes underneath the record format.
func NewSM4GCM(key []byte) (cipher.AEAD, error) {
        if len(key) != SM4KeySize {
                return nil, ErrInvalidLength
        }
        block, err := sm4.NewCipher(key)
        if err != nil {
                return nil, ErrInvalidKey
        }
        aead, err := cipher.NewGCM(block)
        if err != nil {
                return nil, err
        }
        if aead.NonceSize() != SM4NonceSize || aead.Overhead() != SM4TagSize {
                return nil, ErrInvalidLength
        }
        return aead, nil
}

// EventIndex is bounded, rejects malformed/duplicate business facts, and binds
// every accepted/sent/confirmed fact to the FULL identity/session/operation key.
type EventIndex map[string]map[string]BusinessEvent

func ParseEvents(b []byte, run string) (EventIndex, error) {
        index := EventIndex{}
        for _, line := range bytes.Split(b, []byte{'\n'}) {
                if len(bytes.TrimSpace(line)) == 0 {
                        continue
                }
                if len(line) > MaxLine {
                        return nil, fmt.Errorf("event exceeds length limit")
                }
                var raw map[string]json.RawMessage
                if e := Decode(line, &raw); e != nil {
                        return nil, e
                }
                var gotRun, stage, outcome string
                if e := json.Unmarshal(raw["run_id"], &gotRun); e != nil || gotRun != run {
                        return nil, fmt.Errorf("event run mismatch")
                }
                if e := json.Unmarshal(raw["stage"], &stage); e != nil {
                        return nil, e
                }
                if e := json.Unmarshal(raw["outcome"], &outcome); e != nil {
                        return nil, e
                }
                if stage != "business" || outcome != "telemetry-accepted" && outcome != "control-sent" && outcome != "control-confirmed" {
                        continue
                }
                var ev BusinessEvent
                for key, dst := range map[string]any{"run_id": &ev.RunID, "peer_id": &ev.PeerID, "session_id": &ev.SessionID, "sample_id": &ev.SampleID, "operation_id": &ev.OperationID} {
                        if key == "operation_id" && outcome == "telemetry-accepted" {
                                continue
                        }
                        if e := json.Unmarshal(raw[key], dst); e != nil {
                                return nil, fmt.Errorf("event %s: %w", key, e)
                        }
                }
                ev.Stage, ev.Outcome = stage, outcome
                expected := fmt.Sprintf("set-interval-%s-%d", ev.PeerID, ev.SampleID)
                if outcome == "telemetry-accepted" {
                        ev.OperationID = expected
                        if e := json.Unmarshal(raw["plaintext_bytes"], &ev.PlaintextBytes); e != nil || ev.PlaintextBytes <= 0 {
                                return nil, fmt.Errorf("accepted event missing plaintext length")
                        }
                }
                if ev.PeerID == "" || ev.SessionID == "" || ev.SampleID == 0 || ev.OperationID != expected {
                        return nil, fmt.Errorf("invalid business correlation")
                }
                k := ev.Key()
                if index[k] == nil {
                        index[k] = map[string]BusinessEvent{}
                }
                if _, exists := index[k][outcome]; exists {
                        return nil, fmt.Errorf("duplicate business event %s %s", k, outcome)
                }
                index[k][outcome] = ev
                if len(index) > 10000 {
                        return nil, fmt.Errorf("too many business events")
                }
        }
        return index, nil
}
func LoadEvents(path, run string) (EventIndex, error) {
        b, e := ReadBytes(path)
        if e != nil {
                return nil, e
        }
        return ParseEvents(b, run)
}

// ConfirmationObserver is a local measurement side channel, NOT a protocol
// message. It observes the gateway's authenticated ACK validation event on the
// worker's monotonic clock. Costs of event visibility/polling are in the window.
// No H1-H4, KDF, application record or cryptographic semantics are changed.
type ConfirmationObserver struct {
        Path    string
        RunID   string
        Offset  int64
        Pending []byte
        Index   EventIndex
}

func (o *ConfirmationObserver) Wait(ctx context.Context, t Transaction) error {
        if o.Index == nil {
                o.Index = EventIndex{}
        }
        tick := time.NewTicker(time.Millisecond)
        defer tick.Stop()
        for {
                if _, ok := o.Index[t.Key()]["control-confirmed"]; ok {
                        return nil
                }
                f, e := os.Open(o.Path)
                if e != nil {
                        return e
                }
                _, e = f.Seek(o.Offset, io.SeekStart)
                var b []byte
                if e == nil {
                        b, e = io.ReadAll(io.LimitReader(f, MaxFile+1))
                }
                ce := f.Close()
                if e != nil {
                        return e
                }
                if ce != nil {
                        return ce
                }
                if len(b) > MaxFile {
                        return fmt.Errorf("event stream exceeds limit")
                }
                o.Offset += int64(len(b))
                o.Pending = append(o.Pending, b...)
                end := bytes.LastIndexByte(o.Pending, '\n')
                if end >= 0 {
                        part, e := ParseEvents(o.Pending[:end+1], o.RunID)
                        if e != nil {
                                return e
                        }
                        for k, events := range part {
                                if o.Index[k] == nil {
                                        o.Index[k] = map[string]BusinessEvent{}
                                }
                                for outcome, ev := range events {
                                        if _, dup := o.Index[k][outcome]; dup {
                                                return fmt.Errorf("duplicate event")
                                        }
                                        o.Index[k][outcome] = ev
                                }
                        }
                        o.Pending = append([]byte(nil), o.Pending[end+1:]...)
                }
                if len(o.Pending) > MaxLine || len(o.Index) > 10000 {
                        return fmt.Errorf("event observer limit")
                }
                if _, ok := o.Index[t.Key()]["control-confirmed"]; ok {
                        return nil
                }
                select {
                case <-ctx.Done():
                        return ctx.Err()
                case <-tick.C:
                }
        }
}

// CheckFormal is called before provisioning or creating evidence. Tests may
// use interrupted prefixes, but the production requested counts never shrink.
func CheckFormal(kind string, trials, rounds, per int, timeout time.Duration, source, run string) error {
        b, err := hex.DecodeString(source)

                }
        case "throughput":
                if rounds != FormalRounds || per != FormalPerRound {
                        return fmt.Errorf("formal throughput requires 5 rounds of 200 transactions")
                }
        default:
                return fmt.Errorf("unknown formal experiment %q", kind)
        }
        return nil
}

const MaxFile = 64 << 20
const MaxLine = 64 << 10

// A round contains up to 1000 transaction facts; event lines keep MaxLine.
const MaxRow = 2 << 20

// Decode rejects duplicate keys at EVERY nesting level, trailing JSON, absent
// non-optional struct fields and unknown fields. Map users specify their own
// required event fields, because production events have optional metadata.
func Decode(b []byte, dst any) error {
        t := reflect.TypeOf(dst)
        if t == nil || t.Kind() != reflect.Pointer || reflect.ValueOf(dst).IsNil() {
                return errors.New("JSON destination must be a non-nil pointer")
        }
        if len(b) > MaxFile {
                return fmt.Errorf("JSON exceeds size limit")
        }
        if !utf8.Valid(b) {
                return errors.New("JSON contains invalid UTF-8")
        }
        d := json.NewDecoder(bytes.NewReader(b))
        d.UseNumber()
        var walk func() error
        walk = func() error {
                tok, err := d.Token()
                if err != nil {
                        return err
                }
                if delim, ok := tok.(json.Delim); ok {
                        switch delim {
                        case '{':
                                seen := map[string]bool{}
                                for d.More() {
                                        k, e := d.Token()
                                        if e != nil {
                                                return e
                                        }
                                        key := k.(string)
                                        if seen[key] {
                                                return fmt.Errorf("duplicate key %q", key)
                                        }
                                        seen[key] = true
                                        if e = walk(); e != nil {
                                                return e
                                        }
                                }
                        case '[':
                                for d.More() {
                                        if e := walk(); e != nil {
                                                return e
                                        }
                                }
                        default:
                                return fmt.Errorf("unexpected delimiter")
                        }
                        _, err = d.Token()
                        return err
                }
                return nil
        }
        if err := walk(); err != nil {
                return err
        }
        if _, err := d.Token(); err != io.EOF {
                return fmt.Errorf("trailing JSON")
        }
        var raw any
        d = json.NewDecoder(bytes.NewReader(b))
        d.UseNumber()
        if err := d.Decode(&raw); err != nil {
                return err
        }
        if err := presence(raw, reflect.TypeOf(dst).Elem()); err != nil {
                return err
        }
        d = json.NewDecoder(bytes.NewReader(b))
        d.DisallowUnknownFields()
        d.UseNumber()
        return d.Decode(dst)
}

func presence(v any, t reflect.Type) error {
        if t.Kind() == reflect.Pointer {
                if v == nil {
                        return nil
                }
                return presence(v, t.Elem())
        }
        if t.Kind() == reflect.Struct {
                // Custom JSON scalars such as time.Time still need the enclosing field
                // presence check, while their own typed decoder validates the encoding.
                if reflect.PointerTo(t).Implements(reflect.TypeOf((*json.Unmarshaler)(nil)).Elem()) {
                        return nil
                }
                obj, ok := v.(map[string]any)
                if !ok {
                        return fmt.Errorf("expected object")
                }
                for i := 0; i < t.NumField(); i++ {
                        f := t.Field(i)
                        if !f.IsExported() {
                                continue
                        }
                        tag := f.Tag.Get("json")
                        parts := strings.Split(tag, ",")
                        key := parts[0]
                        if key == "-" {
                                continue
                        }
                        if key == "" {
                                key = f.Name
                        }
                        val, exists := obj[key]
                        optional := strings.Contains(tag, ",omitempty")
                        if !exists {
                                if !optional {
                                        return fmt.Errorf("missing field %s", key)
                                }
                                continue
                        }
                        if val == nil && f.Type.Kind() != reflect.Pointer && f.Type.Kind() != reflect.Slice && f.Type.Kind() != reflect.Map && f.Type.Kind() != reflect.Interface {
                                return fmt.Errorf("null field %s", key)
                        }
                        if err := presence(val, f.Type); err != nil {
                                return fmt.Errorf("%s: %w", key, err)
                        }
                }
        }
        if t.Kind() == reflect.Slice {
                if arr, ok := v.([]any); ok {
                        for _, el := range arr {
                                if err := presence(el, t.Elem()); err != nil {
                                        return err
                                }
                        }
                }
        }
        return nil
}

func Read(path string, dst any) error {
        b, e := ReadBytes(path)
        if e != nil {
                return e
        }
        return Decode(b, dst)
}
func ReadBytes(path string) ([]byte, error) {
        f, e := os.Open(path)
        if e != nil {
                return nil, e
        }
        b, e := io.ReadAll(io.LimitReader(f, MaxFile+1))
        closeErr := f.Close()
        if len(b) > MaxFile {
                return nil, errors.Join(e, closeErr, fmt.Errorf("%s exceeds size limit", path))
        }
        return b, errors.Join(e, closeErr)
}
func ReadLines[T any](path string) ([]T, error) {
        b, e := ReadBytes(path)
        if e != nil {
                return nil, e
        }
        s := bufio.NewScanner(bytes.NewReader(b))
        s.Buffer(make([]byte, 4096), MaxRow)
        var rows []T
        for s.Scan() {
                if len(bytes.TrimSpace(s.Bytes())) == 0 {
                        continue
                }
                var row T
                if e = Decode(s.Bytes(), &row); e != nil {
                        return nil, fmt.Errorf("%s row %d: %w", path, len(rows)+1, e)
                }
                rows = append(rows, row)
        }
        return rows, s.Err()
}

func WriteBytes(path string, b []byte) error {
        f, e := os.Create(path)
        if e != nil {
                return e
        }
        return publishFile(f, b)
}

type durableFile interface {
        Write([]byte) (int, error)
        Sync() error
        Close() error
}

func publishFile(f durableFile, b []byte) error {
        n, werr := f.Write(b)
        if werr == nil && n != len(b) {
                werr = io.ErrShortWrite
        }
        return errors.Join(werr, f.Sync(), f.Close())
}
func WriteJSON(path string, v any) error {
        b, e := json.MarshalIndent(v, "", "  ")
        if e != nil {
                return e
        }
        return WriteBytes(path, append(b, '\n'))
}
func WriteLines(path string, v any) error {
        rv := reflect.ValueOf(v)
        if rv.Kind() != reflect.Slice {
                return fmt.Errorf("expected slice")
        }
        var b bytes.Buffer
        enc := json.NewEncoder(&b)
        for i := 0; i < rv.Len(); i++ {
                if e := enc.Encode(rv.Index(i).Interface()); e != nil {
                        return e
                }
        }
        return WriteBytes(path, b.Bytes())
}

func (m Manifest) Hash() string {
        m.ConfigHash = ""
        b, _ := json.Marshal(m)
        h := sha256.Sum256(b)
        return hex.EncodeToString(h[:])
}
func (m Manifest) Check(schema, kind string) error {
        formal := m.RunKind == "formal"
        if formal {
                schema = FormalSchema(kind)
        }
        if m.SchemaVersion != schema || m.Kind != kind || m.RunID == "" || (m.RunKind != "smoke" && !formal) || m.Formal != formal || m.TimeoutNS <= 0 || m.ConfigHash != m.Hash() {
                return fmt.Errorf("invalid manifest schema/run/config/smoke contract")
        }
        if formal {
                if m.PlanID != FormalPlanID {
                        return fmt.Errorf("formal plan mismatch")
                }
                return CheckFormal(kind, m.Trials, m.Rounds, m.PerRound, time.Duration(m.TimeoutNS), m.SourceHash, m.RunID)
        }
        if m.PlanID != "" {
                return fmt.Errorf("smoke must not claim a formal plan")
        }
        if m.SourceHash != "" {
                b, e := hex.DecodeString(m.SourceHash)
                if e != nil || len(b) != 32 {
                        return fmt.Errorf("invalid source hash")
                }
        }
        if (kind == "resource" || kind == "handshake") && (m.Trials < 1 || m.Trials > 100) {
                return fmt.Errorf("invalid smoke trial count")
        }
        if kind == "throughput" && (m.Rounds < 1 || m.Rounds > 10 || m.PerRound < 1 || m.PerRound > 1000) {
                return fmt.Errorf("invalid smoke rounds")
        }
        return nil
}

                return e
        }
        return StatusFile(dir, m.SchemaVersion, m.RunID, "RUNNING", 0, "started")
}

func CheckStatus(dir string, m Manifest, code int, success bool) error {
        var s Status
        if e := Read(filepath.Join(dir, "status.json"), &s); e != nil {
                return e
        }
        if s.SchemaVersion != m.SchemaVersion || s.RunID != m.RunID || s.ExitCode != code {
                return fmt.Errorf("status/run/actual exit mismatch")
        }
        if s.State != "PASS" && s.State != "FAIL" && s.State != "INCOMPLETE" {
                return fmt.Errorf("invalid terminal status")
        }
        if (s.State == "PASS") != (code == 0) || (s.State == "PASS") != success {
                return fmt.Errorf("PASS inconsistent with recomputed run")
        }
        return nil
}

// Complete publishes the producing code, then invokes the COMPLETE validator
// on those same files. validation.json never hides a rejected result.
func Complete(dir, schema, run string, code int, why string, validate func(string, int) []string) (int, error) {
        state := "PASS"
        if code != 0 {
                state = "FAIL"
        }
        if code == 4 {
                state = "INCOMPLETE"
        }
        if e := StatusFile(dir, schema, run, state, code, why); e != nil {
                return 5, e
        }
        problems := validate(dir, code)
        es := "VALID"
        if len(problems) > 0 {
                es = "INVALID"
        }
        if e := WriteJSON(filepath.Join(dir, "validation.json"), map[string]any{"problems": problems, "evidence_status": es, "run_status": state, "producer_exit_code": code, "acceptance_passed": code == 0 && len(problems) == 0}); e != nil {
                se := StatusFile(dir, schema, run, "FAIL", 5, "validation publication failed: "+e.Error())
                if se != nil {
                        return 5, fmt.Errorf("validation publication: %v; final status: %w", e, se)
                }
                return 5, e
        }
        if len(problems) > 0 {
                if e := StatusFile(dir, schema, run, "FAIL", 5, "complete validator rejected output"); e != nil {
                        return 5, e
                }
                return 5, fmt.Errorf("result validation rejected: %v", problems)
        }
        return code, nil
}

func RecomputeThroughput(m Manifest, rows []ThroughputRound) ThroughputSummary {
        s := ThroughputSummary{SchemaVersion: m.SchemaVersion, RunID: m.RunID, Mode: m.Mode, Rounds: m.Rounds, PerRound: m.PerRound, NotEligibleReason: "smoke run: whole business implementation including local confirmation observation and logging; not a single-algorithm or line-rate result"}
        if m.Formal {
                s.FormalResultsEligible = true
                s.NotEligibleReason = ""
        }
        var g, t float64
        n := 0
        for _, r := range rows {
                s.TotalAttempted += r.Attempted
                s.TotalCompleted += r.CompletedTransactions
                s.TotalFailed += r.FailedTransactions
                s.TotalGoodputBytes += r.AcceptedTelemetryPlaintextBytes
                if r.ApplicationGoodputBytesPerSec != nil && r.TransactionsPerSec != nil {
                        g += *r.ApplicationGoodputBytesPerSec
                        t += *r.TransactionsPerSec
                        n++
                }
        }
        if n > 0 {
                g /= float64(n)
                t /= float64(n)
                s.AvgApplicationGoodputBytesPerSec = &g
                s.AvgTransactionsPerSec = &t
        } else {
                s.RateUnavailableReason = "no positive confirmed observation window"
        }
        return s
}

func ValidateThroughput(dir string, code int) []string {
        problems := []string{}
        add := func(e error) {
                if e != nil {
                        problems = append(problems, e.Error())
                }
        }
        var m Manifest
        if e := Read(filepath.Join(dir, "manifest.json"), &m); e != nil {
                return []string{e.Error()}
        }
        add(m.Check("m5b-throughput/2", "throughput"))
        if m.Mode != "traditional" && m.Mode != "hybrid" && m.Mode != "public-hybrid-P" {
                add(fmt.Errorf("invalid throughput mode"))
        }
        rf, sf, ef := ThroughputFiles(m)
        rows, e := ReadLines[ThroughputRound](filepath.Join(dir, rf))
        add(e)
        index, e := LoadEvents(filepath.Join(dir, ef), m.RunID)
        add(e)
        if !(m.Formal && code == 4 && len(rows) <= m.Rounds) && (len(rows) != m.Rounds || len(rows) == 0) {
                add(fmt.Errorf("requested rounds missing"))
        }
        problems = append(problems, CheckThroughputRows(m, rows, index)...)
        var sum ThroughputSummary
        add(Read(filepath.Join(dir, sf), &sum))
        want := RecomputeThroughput(m, rows)
        if !reflect.DeepEqual(sum, want) {
                add(fmt.Errorf("summary does not recompute from raw rounds"))
        }
        success := len(rows) == m.Rounds && want.TotalCompleted == m.Rounds*m.PerRound && want.TotalFailed == 0
        for _, r := range rows {
                if r.ErrorStage != "" || r.ServerError != "" {
                        success = false
                }
        }
        add(CheckStatus(dir, m, code, success))
        return problems
}

func CheckThroughputRows(m Manifest, rows []ThroughputRound, index EventIndex) []string {
        problems := []string{}
        add := func(e error) {
                if e != nil {
                        problems = append(problems, e.Error())
                }
        }
        roundIDs, sessions, keys := map[int]bool{}, map[string]bool{}, map[string]bool{}
        for _, r := range rows {
                if roundIDs[r.Round] {
                        add(fmt.Errorf("round %d appears more than once", r.Round))
                }
                roundIDs[r.Round] = true
                if r.SchemaVersion != "m5b-throughput/2" || r.RunID != m.RunID || r.Mode != m.Mode || r.Round < 1 || r.Round > m.Rounds || r.Attempted != m.PerRound || r.PeerID != "D001" || r.FormalResultsEligible != m.Formal || (m.Formal && r.NotEligibleReason != "") || (!m.Formal && r.NotEligibleReason == "") || r.ConfirmationMethod != ConfirmationMethod {
                        add(fmt.Errorf("round %d contract/run/request mismatch", r.Round))
                }
                if m.Formal && r.Round != len(roundIDs) {
                        add(fmt.Errorf("formal rounds must be a continuous prefix"))
                }
                if m.Formal {
                        if _, e := time.Parse(time.RFC3339Nano, r.StartedAtUTC); e != nil {
                                add(fmt.Errorf("formal round start timestamp missing/invalid"))
                        }
                }
                if r.SessionID != "" {
                        if sessions[r.SessionID] {
                                add(fmt.Errorf("session reused"))
                        }
                        sessions[r.SessionID] = true
                }
                if m.Mode == "public-hybrid-P" {
                        if r.CompletedTransactions > 0 || (!m.Formal && (r.ClientTLS != nil || r.ServerTLS != nil)) || (m.Formal && r.ErrorStage == "") {
                                add(CheckTLS(r.ClientTLS, "gateway.edgehybrid.test"))
                                add(CheckTLS(r.ServerTLS, "D001"))
                        }
                } else if r.ClientTLS != nil || r.ServerTLS != nil {
                        add(fmt.Errorf("T/H row must not claim TLS facts"))
                }
                accepted, sent, confirmed, observed := 0, 0, 0, 0
                var acceptedBytes, lastConfirm, lastAccept, lastAck int64
                if len(r.Transactions) > r.Attempted {
                        add(fmt.Errorf("too many transactions"))
                }
                for i, t := range r.Transactions {
                        if t.RunID != m.RunID || t.PeerID != r.PeerID || t.SessionID != r.SessionID || t.SampleID != uint64(r.Round*m.PerRound+i+1) || t.OperationID != fmt.Sprintf("set-interval-%s-%d", t.PeerID, t.SampleID) || t.PlaintextBytes <= 0 {
                                add(fmt.Errorf("transaction key/sequence/length mismatch"))
                        }
                        k := t.Key()
                        if keys[k] {
                                add(fmt.Errorf("duplicate transaction ID"))
                        }
                        keys[k] = true
                        facts := index[k]
                        a, aok := facts["telemetry-accepted"]
                        _, sok := facts["control-sent"]
                        _, cok := facts["control-confirmed"]
                        if aok {
                                accepted++
                                acceptedBytes += a.PlaintextBytes
                                if a.PlaintextBytes != t.PlaintextBytes {
                                        add(fmt.Errorf("accepted plaintext length mismatch"))
                                }
                        }
                        if sok {
                                sent++
                                if !aok {
                                        add(fmt.Errorf("sent without accepted"))
                                }
                        }
                        if cok {
                                confirmed++
                                if !sok || !aok {
                                        add(fmt.Errorf("confirmation lacks preceding facts"))
                                }
                        }
                        if t.ConfirmedObserved {
                                if !cok || !t.AckSent || !t.AcceptedObserved || t.ConfirmedObservedNS < t.AckSentNS || t.AckSentNS < t.AcceptedObservedNS {
                                        add(fmt.Errorf("confirmed observation without matching remote ACK event/timing"))
                                }
                                observed++
                                lastConfirm = t.ConfirmedObservedNS
                        }
                        if t.ConfirmedObservedNS < 0 || t.AckSentNS < 0 || t.AcceptedObservedNS < 0 || !t.ConfirmedObserved && t.ConfirmedObservedNS != 0 || !t.AckSent && t.AckSentNS != 0 || !t.AcceptedObserved && t.AcceptedObservedNS != 0 {
                                add(fmt.Errorf("negative transaction timing"))
                        }
                        if t.AcceptedObserved {
                                lastAccept = t.AcceptedObservedNS
                        }
                        if t.AckSent {
                                lastAck = t.AckSentNS
                        }
                }
                if r.AcceptedCount != accepted || r.GatewayAcceptedCount != accepted || r.GatewaySentCount != sent || r.GatewayConfirmedCount != confirmed || r.CompletedTransactions != observed || r.FailedTransactions != r.Attempted-observed || r.AcceptedTelemetryPlaintextBytes != acceptedBytes {
                        add(fmt.Errorf("round counts/bytes do not recompute from real accepted/sent/confirmed events"))
                }
                if r.AcceptConfirmedWindowNS != lastAccept || r.AckSendWindowNS != lastAck || r.EndToEndConfirmedWindowNS != lastConfirm {
                        add(fmt.Errorf("round measurement windows do not recompute"))
                }
                gps, tps, reason := Rates(observed, acceptedBytes, lastConfirm)
                if !reflect.DeepEqual(r.ApplicationGoodputBytesPerSec, gps) || !reflect.DeepEqual(r.TransactionsPerSec, tps) || r.RateUnavailableReason != reason {
                        add(fmt.Errorf("round rates do not recompute"))
                }
                if r.WireC2SBase < 0 || r.WireS2CBase < 0 || r.WireC2SEnd < r.WireC2SBase || r.WireS2CEnd < r.WireS2CBase || r.WireClientToServerBytes != r.WireC2SEnd-r.WireC2SBase || r.WireServerToClientBytes != r.WireS2CEnd-r.WireS2CBase {
                        add(fmt.Errorf("wire phase differences do not recompute"))
                }
                if m.Mode == "public-hybrid-P" {
                        if r.Suite != 0 || r.ClientTLS != nil && r.AuthenticatedPeerID != "gateway.edgehybrid.test" {
                                add(fmt.Errorf("P peer/suite mismatch"))
                        }
                } else {
                        want := uint16(1)
                        if m.Mode == "hybrid" {
                                want = 2
                        }
                        if r.SessionID != "" && (r.Suite != want || r.AuthenticatedPeerID != "GW001") {
                                add(fmt.Errorf("T/H authenticated peer/suite mismatch"))
                        }
                }
                if r.WireClientToServerBytes < 0 || r.WireServerToClientBytes < 0 || confirmed > 0 && (r.WireClientToServerBytes == 0 || r.WireServerToClientBytes == 0) {
                        add(fmt.Errorf("invalid business wire differences"))
                }
                if r.GatewayReceiveWindowNS == nil && r.GatewayWindowReason == "" {
                        add(fmt.Errorf("missing gateway window value/reason"))
                }
                if r.FailedTransactions > 0 && r.ErrorStage == "" {
                        add(fmt.Errorf("failed round missing error stage"))
                }
        }
        for k := range index {
                if !keys[k] {
                        add(fmt.Errorf("unmatched gateway business event %s", k))
                }
        }
        return problems
}

func Rates(completed int, bytes, ns int64) (*float64, *float64, string) {
        if ns <= 0 {
                return nil, nil, "no positive confirmed observation window"
        }
        g, t := float64(bytes)*1e9/float64(ns), float64(completed)*1e9/float64(ns)
        return &g, &t, ""
}

// Channel is an authenticated, encrypted, replay-resistant byte stream between
// one device and one gateway.
//
// A Channel is only ever produced after both signatures, both Finished values
// and the final transcript have been verified. There is no mock Channel and no
// constructor that skips verification.
type Channel interface {
        // Send seals one application record and writes it. The sequence number is
        // allocated and consumed under a single lock that also covers the network
        // write, so concurrent Send calls preserve TCP ordering. Waiting for that
        // lock is cancellable, and cancellation during the write closes the session.
        Send(ctx context.Context, contentType byte, plaintext []byte) error
        // Receive returns the next authenticated record's content type and
        // plaintext. Receive state advances ONLY after the AEAD tag verifies.
        // Waiting for the receive slot is cancellable.
        Receive(ctx context.Context) (contentType byte, plaintext []byte, err error)
        // SessionID is the public session identifier bound into every record header.
        // It is not secret — it travels in cleartext on every record.
        SessionID() [16]byte
        // PeerID is the AUTHENTICATED identity of the other endpoint. It comes from
        // the signature-verified handshake, not from the business payload, so a
        // service can check that a message's claimed sender matches who really
        // authenticated.
        PeerID() string
        // Suite is the negotiated suite.
        Suite() uint16
        // PeerSnapshot returns the identity material (key ids, generation) that this
        // handshake actually verified, so a service can compare it with the current
        // registry before admitting business traffic.
        PeerSnapshot() PeerSnapshot
        Close() error
}

type secureChannel struct {
        conn        net.Conn
        maxFrame    uint32
        timeout     time.Duration
        idleTimeout time.Duration

        send *record.Sender
        recv *record.Receiver
        sid  [16]byte
        peer string
        suit uint16

        // Cancellable serialisation. A caller blocked here can be cancelled, which a
        // sync.Mutex would not allow.
        sendSem semaphore
        recvSem semaphore
        // Per-direction generation counters: a stale cancellation callback must not
        // expire the deadline of a later operation.
        sendGuard ioGuard
        recvGuard ioGuard

        // peerSnapshot is the identity material this handshake actually verified.
        peerSnapshot PeerSnapshot

        closeOnce sync.Once
        closeErr  error
        closed    chan struct{}
}

// PeerSnapshot is the identity material verified by one handshake.
//
// It is recorded at authentication time and never updated, so the service can
// compare it with the CURRENT registry later. An empty SignKeyID means the
// handshake did not record one (for example a peer that was not resolved through
// the registry), and a comparison must then be treated as "unknown", not as a
// match.
type PeerSnapshot struct {
        PeerID           string
        SignKeyID        string
        AgreementKeyID   string
        KeyGeneration    uint64
        RegistryRevision uint64
}

// Matches reports whether the snapshot still agrees with a freshly read registry
// entry.
//
// EVERY field is compared, including RegistryRevision. The revision is the
// PER-ENTRY revision of the peer record (not the whole-registry revision), so an
// unrelated device's update cannot invalidate this comparison -- but a policy or
// key change on THIS device between the handshake and admission does, and that
// is exactly what admission must catch. A snapshot with no recorded key ids
// cannot be compared and reports false, so a caller fails closed rather than
// assuming equality.
func (s PeerSnapshot) Matches(peerID, signKeyID, agreementKeyID string, generation, revision uint64) bool {
        if s.PeerID != "" && s.PeerID != peerID {
                return false
        }
        if s.SignKeyID == "" || s.AgreementKeyID == "" {
                return false
        }
        return s.SignKeyID == signKeyID && s.AgreementKeyID == agreementKeyID &&
                s.KeyGeneration == generation && s.RegistryRevision == revision
}

func newChannel(conn net.Conn, maxFrame uint32, timeout time.Duration, sm sessionMaterial, suite uint16, peerID string, snap PeerSnapshot, isInitiator bool, limit uint64, budget int) (Channel, error) {
        if snap.PeerID == "" {
                snap.PeerID = peerID
        }
        meta := func(dir byte) record.Meta {
                return record.Meta{Suite: suite, SessionID: sm.sid, Epoch: 1, Direction: dir}
        }
        var sendKey, recvKey []byte
        var sendPrefix, recvPrefix [4]byte
        var sendDir, recvDir byte
        if isInitiator {
                sendKey, sendPrefix, sendDir = sm.kAtoB[:], sm.pAtoB, record.DeviceToGateway
                recvKey, recvPrefix, recvDir = sm.kBtoA[:], sm.pBtoA, record.GatewayToDevice
        } else {
                sendKey, sendPrefix, sendDir = sm.kBtoA[:], sm.pBtoA, record.GatewayToDevice
                recvKey, recvPrefix, recvDir = sm.kAtoB[:], sm.pAtoB, record.DeviceToGateway
        }
        // Real SM4-GCM, from the vetted adapter. Not an AES test double.
        sendAEAD, err := cryptosuite.NewSM4GCM(sendKey)
        if err != nil {
                return nil, err
        }
        recvAEAD, err := cryptosuite.NewSM4GCM(recvKey)
        if err != nil {
                return nil, err
        }
        sender, err := record.NewSenderBudget(sendAEAD, meta(sendDir), sendPrefix, limit, budget)
        if err != nil {
                return nil, err
        }
        receiver, err := record.NewReceiverBudget(recvAEAD, meta(recvDir), recvPrefix, limit, budget)
        if err != nil {
                return nil, err
        }
        return &secureChannel{
                conn: conn, maxFrame: maxFrame, timeout: timeout,
                send: sender, recv: receiver, sid: sm.sid, peer: peerID, suit: suite,
                peerSnapshot: snap,
                sendSem:      newSemaphore(), recvSem: newSemaphore(),
                closed: make(chan struct{}),
        }, nil
}

func (c *secureChannel) SessionID() [16]byte { return c.sid }
func (c *secureChannel) PeerID() string      { return c.peer }
func (c *secureChannel) Suite() uint16       { return c.suit }

// PeerSnapshot returns the identity material that THIS HANDSHAKE actually
// verified: the peer's signing key id, agreement key id and key generation, as
// they stood at the moment the signature and the Finished message were checked.
//
// Why it matters: a handshake authenticates against a specific generation of a
// specific key pair. If the registry is rotated while a handshake is in flight,
// the session was still authenticated with the OLD keys. Recording what was
// actually verified lets the service compare against the LATEST registry at
// business-admission time instead of assuming they are the same -- so a session
// authenticated with generation N is never registered as generation N+1.
func (c *secureChannel) PeerSnapshot() PeerSnapshot { return c.peerSnapshot }

func (c *secureChannel) isClosed() bool {
        select {
        case <-c.closed:
                return true
        default:
                return false
        }
}

// Close is idempotent and never blocks on the send or receive lock, so a peer
// that stops reading cannot deadlock shutdown. Closing the connection also
// unblocks any Read or Write in flight.
func (c *secureChannel) Close() error {
        c.closeOnce.Do(func() {
                close(c.closed)
                c.closeErr = c.conn.Close()
        })
        return c.closeErr
}

func (c *secureChannel) Send(ctx context.Context, contentType byte, plaintext []byte) error {
        if c.isClosed() {
                return ErrClosed
        }
        // Waiting for the send slot is cancellable. A waiter cancelled here has sent
        // nothing and consumed nothing, so the session stays usable.
        if err := c.sendSem.acquire(ctx); err != nil {
                return err
        }
        defer c.sendSem.release()

        if c.isClosed() {
                return ErrClosed
        }
        // Cancelled after acquiring but before consuming a sequence number: nothing
        // has happened yet, so the session is still valid.
        if err := cancelError(ctx); err != nil {
                return err
        }

        stop := armIO(ctx, writeDeadlineSetter(c.conn), c.timeout, time.Time{}, &c.sendGuard)
        defer stop()

        sealed, err := c.send.Seal(contentType, plaintext)
        if err != nil {
                // The sequence number has been consumed. Continuing would risk reusing
                // it under the same key, so the session ends instead.
                _ = c.Close()
                return err
        }
        enc, err := wire.Encode(wire.Frame{Kind: wire.ProtectedRecord, Payload: sealed}, c.maxFrame)
        if err != nil {
                _ = c.Close()
                return err
        }
        if err := wire.WriteEncoded(c.conn, enc); err != nil {
                // A partial or failed write leaves the stream in an unknown state, and a
                // sequence number has already been consumed. The only safe response is to
                // close; the sequence is NOT rolled back and the Sender is never rebuilt
                // with the old key.
                _ = c.Close()
                // Classify by cause: a cancellation stays context.Canceled, the caller's
                // own deadline becomes context.DeadlineExceeded, a local timeout becomes
                // ErrTimeout, and a transport failure keeps its own error.
                if classified := operationError(ctx, err); classified != err {
                        return classified
                }
                return fmt.Errorf("%w: record write failed: %v", ErrProtocol, err)
        }
        return nil
}

func (c *secureChannel) Receive(ctx context.Context) (byte, []byte, error) {
        if c.isClosed() {
                return 0, nil, ErrClosed
        }
        if err := c.recvSem.acquire(ctx); err != nil {
                return 0, nil, err
        }
        defer c.recvSem.release()

        if c.isClosed() {
                return 0, nil, ErrClosed
        }
        if err := cancelError(ctx); err != nil {
                return 0, nil, err
        }

        timeout := c.timeout
        if c.idleTimeout > 0 {
                timeout = c.idleTimeout
        }
        stop := armIO(ctx, readDeadlineSetter(c.conn), timeout, time.Time{}, &c.recvGuard)
        defer stop()

        reader := io.Reader(c.conn)
        if c.idleTimeout > 0 {
                var first [1]byte
                if _, err := io.ReadFull(c.conn, first[:]); err != nil {
                        _ = c.Close()
                        if classified := operationError(ctx, err); classified != err {
                                return 0, nil, classified
                        }
                        if errors.Is(err, io.EOF) || errors.Is(err, net.ErrClosed) {
                                return 0, nil, ErrPeerClosed
                        }
                        return 0, nil, err
                }
                stop()
                stop = armIO(ctx, readDeadlineSetter(c.conn), c.timeout, time.Time{}, &c.recvGuard)
                defer stop()
                reader = io.MultiReader(bytes.NewReader(first[:]), c.conn)
        }
        f, _, err := wire.ReadRaw(reader, c.maxFrame)
        if err != nil {
                _ = c.Close()
                if classified := operationError(ctx, err); classified != err {
                        return 0, nil, classified
                }
                switch {
                case errors.Is(err, wire.ErrTruncated):
                        // The peer began a frame and the stream ended inside it. That is not
                        // a clean close and must not be reported as one.
                        return 0, nil, fmt.Errorf("%w: %w: %w", ErrProtocol, ErrTruncated, err)
                case errors.Is(err, io.EOF), errors.Is(err, net.ErrClosed):
                        // Nothing of a next frame was received: an ordinary end of
                        // conversation.
                        return 0, nil, ErrPeerClosed
                default:
                        return 0, nil, fmt.Errorf("%w: record read failed: %w", ErrProtocol, err)
                }
        }
        if f.Kind != wire.ProtectedRecord {
                _ = c.Close()
                return 0, nil, fmt.Errorf("%w: expected an application record, received message kind %d", ErrProtocol, f.Kind)
        }
        // record.Receiver advances its sequence ONLY here, after the tag verifies.
        kind, pt, err := c.recv.Open(f.Payload)
        if err != nil {
                _ = c.Close()
                return 0, nil, err
        }
        return kind, pt, nil
}

// ErrRecordLimit is returned when a direction has exhausted its record budget.
// The session must be closed and a fresh full handshake performed; sequence
// numbers are never reset under the same key.
var ErrRecordLimit = record.ErrLimit

func profileChannel(ch Channel, err error, idle time.Duration) (Channel, error) {
        if err == nil && idle > 0 {
                ch.(*secureChannel).idleTimeout = idle
        }
        return ch, err
}

// IsClosed reports whether the channel has been torn down. Tests use it to
// assert that a failure actually terminated the session rather than leaving a
// half-open connection behind.
func IsClosed(ch Channel) bool {
        sc, ok := ch.(*secureChannel)
        if !ok {
                return true
        }
        return sc.isClosed()
}

// Client runs the device side: SM2 initiator, msg1 sender, msg3 sender.
//
// The returned Channel exists only after the gateway's signature AND the
// gateway's Finished have both verified. On any failure the connection is
// closed, so a caller cannot accidentally continue on a half-authenticated
// socket.
func Client(ctx context.Context, c Config) (Channel, error) {
        ch, err := clientHandshake(ctx, c)
        if err != nil {
                // c.Conn may be nil: configuration validation runs before any I/O, and a
                // caller that passed no connection still deserves a clean error rather
                // than a panic on the error path.
                if c.Conn != nil {
                        _ = c.Conn.Close()
                }
                return nil, err
        }
        return ch, nil
}

func clientHandshake(ctx context.Context, c Config) (Channel, error) {
        r, err := c.prepare(identity.RoleDevice)
        if err != nil {
                return nil, err
        }
        if !identity.ValidName(c.ExpectedPeerID) {
                return nil, fmt.Errorf("%w: an expected gateway identity is required", ErrConfig)
        }
        peer, err := r.lookup(ctx, c.ExpectedPeerID, identity.RoleGateway)
        if err != nil {
                return nil, err
        }

        ag := cryptosuite.NewSM2Agreement()
        eph, err := ag.GenerateEphemeral()
        if err != nil {
                return nil, fmt.Errorf("%w: ephemeral SM2 key: %v", ErrProtocol, err)
        }
        defer func() { _ = eph.Private.Destroy() }()
        if len(eph.Public) != SM2PointSize {
                return nil, fmt.Errorf("%w: ephemeral SM2 key has length %d", ErrProtocol, len(eph.Public))
        }

        na, err := randomBytes(NonceSize)
        if err != nil {
                return nil, err
        }

        // ONE absolute deadline for the whole handshake, taken from the earlier of the
        // caller's deadline and the local budget. Messages never extend it.
        fio, err := newFrameIO(ctx, c.Conn, c.MaxFrame, c.Timeout)
        if err != nil {
                return nil, err
        }

        // --- ML-KEM-768 ephemeral key (suite 2 only) ----------------------------
        //
        // Generated per handshake, never cached and never reused: a reused
        // decapsulation key would make every session that used it share one PQ
        // secret. The key is only created when suite 2 is actually OFFERED, and the
        // server enforces the matching presence rule.
        var kemK cryptosuite.KEM
        var kemPriv cryptosuite.PrivateHandle
        var kemPub []byte
        if offersHybrid(r.u16) {
                kemK = c.kem()
                pub, priv, kerr := kemK.Generate()
                if kerr != nil {
                        return nil, fmt.Errorf("%w: generating the ML-KEM ephemeral key: %v", ErrProtocol, kerr)
                }
                kemPub, kemPriv = pub, priv
                defer func() { _ = kemPriv.Destroy() }()
        }

        // --- msg1: ClientHello ---------------------------------------------------
        hello := ClientHelloBody{
                OfferedSuites:       r.u16,
                InitiatorID:         c.Local.ID,
                InitiatorSignKeyID:  c.Local.SignKeyID,
                InitiatorAgreeKeyID: c.Local.AgreeKeyID,
                ResponderID:         c.ExpectedPeerID,
                Nonce:               na,
                EphemeralSM2:        eph.Public,
                EphemeralKEM:        kemPub, // empty unless suite 2 is offered
        }
        if err := checkClientKEMField(r.u16, kemPub); err != nil {
                return nil, err
        }
        m1, err := fio.send(wire.ClientHello, hello.encode())
        if err != nil {
                return nil, err
        }

        // --- msg2: ServerHello + gateway signature -------------------------------
        payload2, m2, err := fio.recv(wire.ServerHello)
        if err != nil {
                return nil, err
        }
        coreBytes, sigB, err := splitServerHello(payload2)
        if err != nil {
                return nil, fmt.Errorf("%w: ServerHello is too short", ErrProtocol)
        }
        core, err := parseServerCore(coreBytes)
        if err != nil {
                return nil, fmt.Errorf("%w: ServerCore: %v", ErrProtocol, err)
        }
        if core.InitiatorID != c.Local.ID || core.ResponderID != peer.Public.ID {
                return nil, fmt.Errorf("%w: ServerHello names initiator %q and responder %q", ErrPeerIdentity, core.InitiatorID, core.ResponderID)
        }
        if core.ResponderSignKeyID != peer.Public.SignKeyID || core.ResponderAgreeKeyID != peer.Public.AgreementKeyID {
                return nil, fmt.Errorf("%w: gateway presented key identifiers the registry does not hold", ErrPeerIdentity)
        }
        // Recompute the selection from the server's advertised list and the TRUSTED
        // minimum modes. If the server picked anything else, the handshake stops
        // here rather than accepting a suite the client never agreed to.
        //
        // This is one of the two structural downgrade defences: the client does not
        // simply accept whatever the server announced. An attacker who deletes the
        // hybrid entry from H1 makes the SERVER select suite 1, but the client
        // recomputes suite 2 from its own configuration and rejects the mismatch.
        sel, err := policy.Select(r.suites, toSuites(core.SupportedSuites), c.MinimumMode, peer.MinimumMode, c.AllowLegacy)
        if err != nil {
                return nil, fmt.Errorf("%w: no mutually acceptable suite: %v", ErrUnsupported, err)
        }
        if uint16(sel) != core.SelectedSuite {
                return nil, fmt.Errorf("%w: gateway selected suite %d but %d was expected", ErrProtocol, core.SelectedSuite, uint16(sel))
        }
        // The ciphertext rule follows what was SELECTED, not what was offered.
        if err := checkServerKEMField(core.SelectedSuite, core.KEMCiphertext); err != nil {
                return nil, err
        }
        // A hybrid selection is only meaningful if this client actually offered it.
        if core.SelectedSuite == SuiteHybrid && !offersHybrid(r.u16) {
                return nil, fmt.Errorf("%w: gateway selected the hybrid suite although it was never offered", ErrProtocol)
        }
        if err := cryptosuite.NewSM2Identity().Verify(
                peer.Public.SignPublic, peer.Public.SM2UserID,
                serverAuthMessage(m1, coreBytes), sigB,
        ); err != nil {
                return nil, fmt.Errorf("%w: gateway signature: %v", ErrAuthentication, err)
        }

        // --- SM2 key exchange and key schedule ----------------------------------
        peerID := cryptosuite.IdentityPublic{
                ID:              peer.Public.ID,
                SM2UserID:       peer.Public.SM2UserID,
                AgreementPublic: peer.Public.AgreementPublic,
        }
        localID := cryptosuite.IdentityPublic{ID: c.Local.ID, SM2UserID: c.Local.SM2UserID}
        if ph, ok := c.Local.AgreeKey.(cryptosuite.PublicHandle); ok {
                if pub, perr := ph.Public(); perr == nil {
                        localID.AgreementPublic = pub
                }
        }
        res, err := ag.Derive(cryptosuite.Initiator, localID, c.Local.AgreeKey, eph, peerID, core.EphemeralSM2)
        if err != nil {
                return nil, fmt.Errorf("%w: SM2 key exchange: %w", ErrProtocol, err)
        }
        defer clear(res.SharedSecret)
        defer clear(res.SendConfirmation)
        defer clear(res.ExpectedPeerConfirmation)
        if len(res.ExpectedPeerConfirmation) != cryptosuite.SM3Size || !cryptosuite.ConstantTimeEqual(core.SM2Confirmation, res.ExpectedPeerConfirmation) {
                return nil, fmt.Errorf("%w: gateway S1", ErrSM2Confirmation)
        }

        // --- ML-KEM-768 decapsulation (suite 2 only) ----------------------------
        //
        // FIPS 203 implicit rejection: a ciphertext of the correct length that is
        // NOT a genuine encryption returns a DIFFERENT pseudorandom secret with
        // err == nil. So a nil error here proves nothing about authenticity; only
        // the Finished values can detect a substituted ciphertext.
        var zpq []byte
        if core.SelectedSuite == SuiteHybrid {
                if kemK == nil || kemPriv == nil {
                        return nil, fmt.Errorf("%w: the hybrid suite was selected but no ML-KEM key was generated", ErrProtocol)
                }
                zpq, err = kemK.Decapsulate(kemPriv, core.KEMCiphertext)
                if err != nil {
                        return nil, fmt.Errorf("%w: ML-KEM decapsulation: %v", ErrProtocol, err)
                }
        }

        kikm, err := ikm(core.SelectedSuite, res.SharedSecret, zpq)
        if err != nil {
                return nil, err
        }
        slt, err := salt(na, core.Nonce)
        if err != nil {
                return nil, err
        }
        prk := cryptosuite.HKDFExtractSM3(slt, kikm)
        t2 := transcriptHashT2(m1, m2)
        kfinA, kfinB, err := finishedKeys(prk, t2)
        if err != nil {
                return nil, err
        }

        // --- msg3: ClientCore + signature + FinishedA ---------------------------
        cc := ClientCoreBody{
                SelectedSuite:       core.SelectedSuite,
                InitiatorID:         c.Local.ID,
                InitiatorSignKeyID:  c.Local.SignKeyID,
                InitiatorAgreeKeyID: c.Local.AgreeKeyID,
                SM2Confirmation:     res.SendConfirmation,
        }
        ccBytes := cc.encode()
        sigA, err := cryptosuite.NewSM2Identity().Sign(
                c.Local.SignKey, c.Local.SM2UserID,
                clientAuthMessage(m1, m2, ccBytes),
        )
        if err != nil {
                return nil, fmt.Errorf("%w: signing ClientAuth: %v", ErrProtocol, err)
        }
        fa := confirmA(kfinA, m1, m2, ccBytes, sigA)
        payload3 := make([]byte, 0, len(ccBytes)+len(sigA)+len(fa))
        payload3 = append(payload3, ccBytes...)
        payload3 = append(payload3, sigA...)
        payload3 = append(payload3, fa...)
        m3, err := fio.send(wire.ClientAuth, payload3)
        if err != nil {
                return nil, err
        }

        // --- msg4: ServerFinished ------------------------------------------------
        payload4, m4, err := fio.recv(wire.ServerFinished)
        if err != nil {
                return nil, err
        }
        if len(payload4) != FinishedSize {
                return nil, fmt.Errorf("%w: FinishedB has length %d", ErrProtocol, len(payload4))
        }
        wantB := confirmB(kfinB, m1, m2, m3)
        if !cryptosuite.ConstantTimeEqual(payload4, wantB) {
                return nil, fmt.Errorf("%w: gateway Finished did not match", ErrTranscript)
        }

        // Only now — after both signatures, both confirmations and the complete
        // transcript — may session keys be derived and a Channel be handed out.
        t4 := transcriptHashT4(m1, m2, m3, m4)
        sm, err := sessionKeys(prk, t4)
        if err != nil {
                return nil, err
        }
        snap := PeerSnapshot{
                PeerID:           peer.Public.ID,
                SignKeyID:        peer.Public.SignKeyID,
                AgreementKeyID:   peer.Public.AgreementKeyID,
                KeyGeneration:    peer.KeyGeneration,
                RegistryRevision: peer.Revision,
        }
        ch, err := newChannel(c.Conn, c.MaxFrame, c.Timeout, sm, core.SelectedSuite, peer.Public.ID, snap, true, r.limit, r.budget)
        return profileChannel(ch, err, c.IdleTimeout)
}

// Package handshake implements the project's frozen traditional handshake
// (suite 1: SM2 key exchange + SM2 signatures + HKDF-SM3 + SM4-GCM) and the
// authenticated record channel that follows it.
//
// This package is a project design, not a standard protocol, and it has not
// been through a formal security proof or an external audit. See
// docs/PROTOCOL_REVIEW_M2.md and docs/PROTOCOL_DRAFT.md.
package handshake

import (
        "encoding/binary"
        "errors"
        "fmt"

        "example.com/edgehybrid/internal/cryptosuite"
        "example.com/edgehybrid/internal/identity"
)

var (
        // ErrCodec means the message bytes were not canonical: wrong length,
        // unknown field, duplicate, or trailing data. It never means "authentication
        // failed" — authentication failures are reported separately so a caller can
        // tell a parse bug from an attack.
        ErrCodec = errors.New("handshake: malformed or non-canonical message")
        // ErrField means a field parsed cleanly but held a value the protocol does
        // not accept (unknown suite, wrong role, illegal name).
        ErrField = errors.New("handshake: field value rejected")
)

const (
        // Body version 2 requires standard SM2 S1/S2. Version 1 is rejected;
        // the outer wire frame / protected record version remains unchanged.
        HandshakeVersion byte = 2
        RoleInitiator    byte = 1
        RoleResponder    byte = 2

        NonceSize    = 32
        MaxNameLen   = 64
        MaxSuites    = 2
        SigSize      = cryptosuite.SM2SignatureSize
        FinishedSize = cryptosuite.SM3Size
        SM2PointSize = cryptosuite.SM2CoordinateSize

        // SuiteTraditional and SuiteHybrid mirror policy.Suite values. They are
        // repeated here so the codec does not have to import policy just to reject
        // an out-of-range identifier.
        SuiteTraditional uint16 = 1
        SuiteHybrid      uint16 = 2
)

func knownSuite(v uint16) bool { return v == SuiteTraditional || v == SuiteHybrid }

// --- canonical writer -------------------------------------------------------

type writer struct{ b []byte }

func (w *writer) u8(v byte) { w.b = append(w.b, v) }
func (w *writer) u16(v uint16) {
        var t [2]byte
        binary.BigEndian.PutUint16(t[:], v)
        w.b = append(w.b, t[:]...)
}
func (w *writer) u32(v uint32) {
        var t [4]byte
        binary.BigEndian.PutUint32(t[:], v)
        w.b = append(w.b, t[:]...)
}
func (w *writer) raw(b []byte) { w.b = append(w.b, b...) }

// vec writes an explicit 4-byte big-endian length before the value, so no two
// different field sequences can ever produce the same bytes.
func (w *writer) vec(b []byte) { w.u32(uint32(len(b))); w.b = append(w.b, b...) }
func (w *writer) str(s string) { w.vec([]byte(s)) }
func (w *writer) suites(s []uint16) {
        w.u16(uint16(len(s)))
        for _, v := range s {
                w.u16(v)
        }
}

// --- strict reader ----------------------------------------------------------

type reader struct {
        b   []byte
        off int
}

func newReader(b []byte) *reader { return &reader{b: b} }

func (r *reader) remaining() int { return len(r.b) - r.off }

func (r *reader) take(n int) ([]byte, error) {
        if n < 0 || r.remaining() < n {
                return nil, ErrCodec
        }
        v := r.b[r.off : r.off+n]
        r.off += n
        return v, nil
}
func (r *reader) u8() (byte, error) {
        v, err := r.take(1)
        if err != nil {
                return 0, err
        }
        return v[0], nil
}
func (r *reader) u16() (uint16, error) {
        v, err := r.take(2)
        if err != nil {
                return 0, err
        }
        return binary.BigEndian.Uint16(v), nil
}
func (r *reader) u32() (uint32, error) {
        v, err := r.take(4)
        if err != nil {
                return 0, err
        }
        return binary.BigEndian.Uint32(v), nil
}
func (r *reader) vec(max int) ([]byte, error) {
        n, err := r.u32()
        if err != nil {
                return nil, err
        }
        if int(n) > max {
                return nil, ErrCodec
        }
        v, err := r.take(int(n))
        if err != nil {
                return nil, err
        }
        return append([]byte(nil), v...), nil
}
func (r *reader) exactVec(n int) ([]byte, error) {
        v, err := r.vec(n)
        if err != nil {
                return nil, err
        }
        if len(v) != n {
                return nil, ErrCodec
        }
        return v, nil
}
func (r *reader) str(max int) (string, error) {
        v, err := r.vec(max)
        if err != nil {
                return "", err
        }
        return string(v), nil
}
func (r *reader) name() (string, error) {
        s, err := r.str(MaxNameLen)
        if err != nil {
                return "", err
        }
        if !identity.ValidName(s) {
                return "", ErrField
        }
        return s, nil
}
func (r *reader) suites() ([]uint16, error) {
        c, err := r.u16()
        if err != nil {
                return nil, err
        }
        if c == 0 || c > MaxSuites {
                return nil, ErrCodec
        }
        out := make([]uint16, 0, c)
        seen := map[uint16]bool{}
        for i := 0; i < int(c); i++ {
                v, err := r.u16()
                if err != nil {
                        return nil, err
                }
                if !knownSuite(v) || seen[v] {
                        return nil, ErrField
                }
                seen[v] = true
                out = append(out, v)
        }
        return out, nil
}

// done rejects trailing bytes. Without this check a peer could append extra
// fields that a lenient parser would ignore while a different implementation
// might honour them — the classic parser-differential bug.
func (r *reader) done() error {
        if r.remaining() != 0 {
                return ErrCodec
        }
        return nil
}

// splitClientAuth splits the msg3 payload into core, signature and Finished.
func splitClientAuth(payload []byte) (core, sig, finished []byte, err error) {
        if len(payload) < SigSize+FinishedSize {
                return nil, nil, nil, ErrCodec
        }
        core = append([]byte(nil), payload[:len(payload)-SigSize-FinishedSize]...)
        sig = append([]byte(nil), payload[len(payload)-SigSize-FinishedSize:len(payload)-FinishedSize]...)
        finished = append([]byte(nil), payload[len(payload)-FinishedSize:]...)
        return core, sig, finished, nil
}

// splitServerHello splits the msg2 payload into ServerCore and SigB.
func splitServerHello(payload []byte) (core, sig []byte, err error) {
        if len(payload) < SigSize {
                return nil, nil, ErrCodec
        }
        core = append([]byte(nil), payload[:len(payload)-SigSize]...)
        sig = append([]byte(nil), payload[len(payload)-SigSize:]...)
        return core, sig, nil
}

// u16b renders a uint16 as a big-endian field for the domain-separated encoder.
func u16b(v uint16) []byte {
        var b [2]byte
        binary.BigEndian.PutUint16(b[:], v)
        return b[:]
}

// --- suite-conditional KEM field rules (M3) --------------------------------
//
// "ADVERTISING" a suite and "SELECTING" a suite are different things, and the
// rules for the KEM fields follow the two separately:
//
//   H1: offering suite 2  -> ekA must be a correctly sized encapsulation key
//       offering only [1] -> ekA must be empty
//
//   H2: selected == 2     -> kem_ct must be a correctly sized ciphertext
//       selected == 1     -> kem_ct must be empty
//
// Both checks are deliberate fail-closed checks. Without them a peer could
// advertise hybrid without a key, or select hybrid without a ciphertext, and
// the session would quietly fall back to depending on SM2 alone.

func offersHybrid(suites []uint16) bool {
        for _, v := range suites {
                if v == SuiteHybrid {
                        return true
                }
        }
        return false
}

// checkClientKEMField validates the H1 encapsulation key against what H1 offers.
func checkClientKEMField(offered []uint16, ek []byte) error {
        if offersHybrid(offered) {
                if len(ek) != cryptosuite.MLKEM768EncapKeySize {
                        return fmt.Errorf("%w: hybrid is offered so the ML-KEM encapsulation key must be %d bytes, got %d",
                                ErrProtocol, cryptosuite.MLKEM768EncapKeySize, len(ek))
                }
                if allZero(ek) {
                        return fmt.Errorf("%w: the ML-KEM encapsulation key is all zeroes", ErrProtocol)
                }
                return nil
        }
        if len(ek) != 0 {
                return fmt.Errorf("%w: only the traditional suite is offered so the ML-KEM encapsulation key must be empty, got %d bytes",
                        ErrProtocol, len(ek))
        }
        return nil
}

// checkServerKEMField validates the H2 ciphertext against the SELECTED suite.
//
// Note the asymmetry with the client check: this follows what was SELECTED,
// not what was advertised, because an H1 that advertised hybrid but ends up
// on suite 1 still leaves ekA in the authenticated transcript.
func checkServerKEMField(selected uint16, ct []byte) error {
        if selected == SuiteHybrid {
                if len(ct) != cryptosuite.MLKEM768CiphertextSize {
                        return fmt.Errorf("%w: suite %d requires a %d-byte ML-KEM ciphertext, got %d",
                                ErrProtocol, SuiteHybrid, cryptosuite.MLKEM768CiphertextSize, len(ct))
                }
                return nil
        }
        if len(ct) != 0 {
                return fmt.Errorf("%w: suite %d must not carry a KEM ciphertext, got %d bytes",
                        ErrProtocol, selected, len(ct))
        }
        return nil
}

// allZero reports whether every byte is zero. This catches only the degenerate
// zero-fill case; it CANNOT detect a reused key from an earlier handshake, and
// the documentation states that limitation rather than implying a guarantee.
func allZero(b []byte) bool {
        for _, v := range b {
                if v != 0 {
                        return false
                }
        }
        return true
}

// Key schedule, exactly as frozen in docs/PROTOCOL_DRAFT.md section 4.
//
// Two properties matter and are easy to get wrong:
//
//  1. The combined helper cryptosuite.HKDFSM3 performs extract-then-expand. Here
//     we must extract ONCE to get the PRK and then expand many times from that
//     same PRK. Calling HKDFSM3 per label would silently re-extract each time
//     and derive keys from a different PRK than the peer, so this file only ever
//     calls HKDFExtractSM3 once and HKDFExpandSM3 afterwards.
//
//  2. No label's input may contain its own output. Finished keys depend on the
//     transcript up to msg2 (T2); session keys depend on the final transcript
//     (T4). Neither depends on itself.

type sessionMaterial struct {
        kAtoB [16]byte // device -> gateway record key
        kBtoA [16]byte // gateway -> device record key
        pAtoB [4]byte  // device -> gateway nonce prefix
        pBtoA [4]byte  // gateway -> device nonce prefix
        sid   [16]byte // session identifier, bound into every record header
}

// ikm builds the extract input for the negotiated suite.
//
// M3 IMPORTANT: in the hybrid suite the input must genuinely depend on BOTH
// secrets. What is forbidden here is not only a literal zero fill, but every
// construction that would make one half irrelevant:
//
//   - using only Zsm2 or only Zpq
//   - XOR-ing the two (each half then contributes a fixed mask, not entropy)
//   - concatenating and truncating to one hash length (drops half the input)
//   - substituting zeroes when the PQ half is missing
//   - reusing a Zpq from an earlier handshake
//
// The encoded form below is domain separated, length prefixed and carries both
// labelled halves, so any change to either secret changes the PRK.
//
// The traditional suite now REJECTS a non-empty PQ half instead of ignoring it.
// Silently ignoring an argument that should not exist would make "does this
// session depend on a KEM secret" depend on whether a caller forgot to pass one;
// rejecting turns that mistake into a detectable failure.
func ikm(suite uint16, zsm2, zpq []byte) ([]byte, error) {
        switch suite {
        case SuiteTraditional:
                if len(zsm2) != cryptosuite.SM3Size {
                        return nil, ErrCodec
                }
                if len(zpq) != 0 {
                        return nil, fmt.Errorf("%w: suite %d must not receive a KEM shared secret (%d bytes); "+
                                "either the suite selection is wrong or a PQ secret leaked into a traditional session",
                                ErrProtocol, SuiteTraditional, len(zpq))
                }
                return cryptosuite.Encode("edgehybrid/v1/ikm", u16b(SuiteTraditional), []byte("sm2"), zsm2), nil
        case SuiteHybrid:
                if len(zsm2) != cryptosuite.SM3Size {
                        return nil, ErrCodec
                }
                if len(zpq) != cryptosuite.MLKEM768SharedSize {
                        // Fail closed: no zero fill, no truncation, no "best effort".
                        return nil, fmt.Errorf("%w: the hybrid suite needs a %d-byte KEM shared secret, got %d",
                                ErrProtocol, cryptosuite.MLKEM768SharedSize, len(zpq))
                }
                if allZero(zpq) {
                        return nil, fmt.Errorf("%w: the KEM shared secret is all zeroes", ErrProtocol)
                }
                return cryptosuite.Encode("edgehybrid/v1/ikm", u16b(SuiteHybrid),
                        []byte("sm2"), zsm2, []byte("mlkem768"), zpq), nil
        default:
                return nil, ErrField
        }
}

// salt = H(E("edgehybrid/v1/salt", Na, Nb)). Both nonces are bound, so neither
// side alone controls the salt.
func salt(na, nb []byte) ([]byte, error) {
        if len(na) != NonceSize || len(nb) != NonceSize {
                return nil, ErrCodec
        }
        return cryptosuite.SumSM3(cryptosuite.Encode("edgehybrid/v1/salt", na, nb)), nil
}

// transcriptHashT2 covers msg1 and msg2 and is the context for both Finished
// keys. It deliberately excludes msg3/msg4 so the Finished keys do not depend on
// the messages they authenticate.
func transcriptHashT2(m1, m2 []byte) []byte {
        return cryptosuite.SumSM3(cryptosuite.Encode("edgehybrid/v1/t2", m1, m2))
}

// transcriptHashT4 covers the complete handshake and is the context for every
// session key. Because it includes msg4, no session key exists until the
// handshake has been fully written.
func transcriptHashT4(m1, m2, m3, m4 []byte) []byte {
        return cryptosuite.SumSM3(cryptosuite.Encode("edgehybrid/v1/t4", m1, m2, m3, m4))
}

// expand derives n bytes from the PRK with an explicit label and context. The
// requested length is encoded into info, so a 16-byte and a 32-byte expansion
// under the same label are different values rather than a truncation of one
// another.
func expand(prk []byte, label string, context []byte, n uint16) ([]byte, error) {
        if len(prk) != cryptosuite.SM3Size {
                return nil, ErrCodec
        }
        info := cryptosuite.Encode(label, context, u16b(n))
        return cryptosuite.HKDFExpandSM3(prk, info, int(n))
}

// finishedKeys derives the two directional Finished keys from the same PRK.
func finishedKeys(prk, t2 []byte) (kfinA, kfinB []byte, err error) {
        if kfinA, err = expand(prk, "edgehybrid/v1/finished-A", t2, cryptosuite.SM3Size); err != nil {
                return nil, nil, err
        }
        if kfinB, err = expand(prk, "edgehybrid/v1/finished-B", t2, cryptosuite.SM3Size); err != nil {
                return nil, nil, err
        }
        return kfinA, kfinB, nil
}

// sessionKeys derives the record keys, nonce prefixes and SID from the PRK and
// the final transcript hash.
func sessionKeys(prk, t4 []byte) (sessionMaterial, error) {
        var m sessionMaterial
        get := func(label string, n uint16) ([]byte, error) {
                return expand(prk, label, t4, n)
        }
        k1, err := get("edgehybrid/v1/app-AtoB", 16)
        if err != nil {
                return m, err
        }
        k2, err := get("edgehybrid/v1/app-BtoA", 16)
        if err != nil {
                return m, err
        }
        p1, err := get("edgehybrid/v1/iv-AtoB", 4)
        if err != nil {
                return m, err
        }
        p2, err := get("edgehybrid/v1/iv-BtoA", 4)
        if err != nil {
                return m, err
        }
        sid, err := get("edgehybrid/v1/session-id", 16)
        if err != nil {
                return m, err
        }
        copy(m.kAtoB[:], k1)
        copy(m.kBtoA[:], k2)
        copy(m.pAtoB[:], p1)
        copy(m.pBtoA[:], p2)
        copy(m.sid[:], sid)
        return m, nil
}

// confirmA computes FinishedA. It covers the client's own core and signature but

}

// Distinct error values so a caller can tell WHY a handshake failed. Reporting
// "handshake failed" for every case would make a tampering attempt look
// identical to a configuration mistake.
var (
        ErrConfig          = errors.New("handshake: invalid local configuration")
        ErrUnsupported     = errors.New("handshake: suite is not implemented in this build")
        ErrPeerIdentity    = errors.New("handshake: peer identity or key identifier rejected")
        ErrAuthentication  = errors.New("handshake: peer signature did not verify")
        ErrTranscript      = errors.New("handshake: Finished confirmation mismatch")
        ErrSM2Confirmation = errors.New("handshake: standard SM2 confirmation mismatch")
        ErrProtocol        = errors.New("handshake: peer violated the message sequence or encoding")
        ErrClosed          = errors.New("handshake: session is closed")
        // ErrPeerClosed means the peer closed the transport cleanly BETWEEN frames.
        // It is a normal end-of-conversation, NOT a protocol violation, and callers
        // should not report it as an attack or a malformed message.
        ErrPeerClosed = errors.New("handshake: peer closed the connection")
        // ErrTruncated means the peer stopped in the MIDDLE of a frame. It is never
        // a clean close: the length prefix had been read, so a frame had begun.
        // Anything that was in flight may be incomplete, so callers must not treat
        // the exchange as finished.
        ErrTruncated = errors.New("handshake: peer stopped in the middle of a frame")
        // ErrTimeout means an I/O deadline expired without the caller's context
        // ending: the local timeout fired, or the peer stalled past the handshake
        // budget. It is kept distinct from ErrProtocol so an operator can tell a
        // slow or dead peer from a malformed message.
        ErrTimeout = errors.New("handshake: I/O deadline exceeded")
)

// LocalIdentity is this endpoint's own key material. The private handles never

// Config carries every dependency explicitly. There is deliberately no package
// level state: two concurrent sessions on one process must not be able to
// influence each other's identity or keys.
type Config struct {
        Conn net.Conn
        // Role is identity.RoleDevice for Client and identity.RoleGateway for Server.
        Role  string
        Local LocalIdentity
        // Suites is the locally declared, ordered capability list. It must contain
        // only suites this build implements, or the endpoint refuses to start.
        Suites      []policy.Suite
        MinimumMode policy.Mode
        AllowLegacy bool
        // ExpectedPeerID is required for Client (the gateway we intend to talk to).
        // Server ignores it: the peer is learned from the ClientHello and then
        // resolved through the trusted registry.
        ExpectedPeerID string
        Trust          identity.TrustStore
        // KEM is the ML-KEM-768 backend used when the hybrid suite is selected.
        // When nil the standard-library adapter is used.
        //
        // Supplying a different implementation does not change the protocol:
        // whatever is supplied must produce and consume real ML-KEM-768 keys,
        // ciphertexts and shared secrets. The point of making it injectable is that
        // an alternative backend (a module, an HSM, or a test double in a _test.go
        // file) can be supplied without any package-global hook in production code.
        KEM cryptosuite.KEM

        MaxFrame uint32
        Timeout  time.Duration
        // Trusted local interactive profile. Zero preserves legacy deadlines.
        // Only waiting for the first byte uses this budget; a partial frame is still
        // bounded by Timeout and never ignored or restarted.
        IdleTimeout time.Duration
        RecordLimit uint64
}

// kem returns the ML-KEM-768 backend for this handshake.
//
// WHY INJECTABLE: the backend is supplied through the configuration rather than
// instantiated inline so that a caller can provide a different ML-KEM-768
// implementation (for example one backed by a module or an HSM) without touching
// this package. It defaults to the standard-library adapter and is never nil.
//
// This also removes the need for any package-global test hook: a test that wants
// to exercise a substituted ciphertext supplies its OWN KEM through the config,
// in a _test.go file, and production code has no knowledge of it.
func (c Config) kem() cryptosuite.KEM {
        if c.KEM != nil {
                return c.KEM
        }
        return cryptosuite.NewMLKEM768()
}

// requireSuite refuses a negotiated suite this build cannot run.
//
// This calls the real backend check directly. There is deliberately no
// package-global variable here that a test could reassign. During M3 the hybrid
// path had to be exercised before the gate could be opened, and the seam used
// then was a package-global override; suite 2 is now open, so that seam is gone
// and production code contains no test hook at all.
func requireSuite(suite uint16) error {
        if err := cryptosuite.RequireImplemented(suite); err != nil {
                return fmt.Errorf("%w: suite %d", ErrUnsupported, suite)
        }
        return nil
}

// ValidateSuites checks a locally declared capability list and refuses any suite
// this build cannot actually run.
//
// A configuration that declares hybrid is REFUSED rather than quietly trimmed.
// Advertising a capability the software cannot deliver would be a false claim,
// and silently negotiating something weaker is precisely the downgrade this
// project must not do.
func ValidateSuites(declared []policy.Suite) ([]policy.Suite, error) {
        if len(declared) == 0 || len(declared) > MaxSuites {
                return nil, fmt.Errorf("%w: %d suites declared", ErrConfig, len(declared))
        }
        seen := map[policy.Suite]bool{}
        for _, s := range declared {
                if s != policy.TraditionalSM && s != policy.HybridSMMLKEM768 {
                        return nil, fmt.Errorf("%w: unknown suite %d", ErrConfig, uint16(s))
                }
                if seen[s] {
                        return nil, fmt.Errorf("%w: duplicate suite %d", ErrConfig, uint16(s))
                }
                seen[s] = true
                if err := cryptosuite.RequireImplemented(uint16(s)); err != nil {
                        return nil, fmt.Errorf("%w: suite %d (%s)", ErrUnsupported, uint16(s), modeOf(s))
                }
        }
        return declared, nil
}

func modeOf(s policy.Suite) string {
        if s == policy.HybridSMMLKEM768 {
                return "hybrid"
        }
        return "traditional"
}

// prepare validates the configuration and derives the per-session limits.
func (c Config) prepare(wantRole string) (resolved, error) {
        var r resolved
        r.cfg = c
        if c.Conn == nil {
                return r, fmt.Errorf("%w: nil connection", ErrConfig)
        }
        if c.Role != wantRole {
                return r, fmt.Errorf("%w: role %q cannot use this entry point", ErrConfig, c.Role)
        }
        if c.MaxFrame < 2048 || c.MaxFrame > wire.HardMaxFrame {
                return r, fmt.Errorf("%w: max frame %d out of range", ErrConfig, c.MaxFrame)
        }
        if c.Timeout <= 0 {
                return r, fmt.Errorf("%w: timeout must be positive", ErrConfig)
        }
        if c.Trust == nil {
                return r, fmt.Errorf("%w: a trusted registry is required", ErrConfig)
        }
        if c.MinimumMode != policy.Traditional && c.MinimumMode != policy.Hybrid {
                return r, fmt.Errorf("%w: invalid minimum mode", ErrConfig)
        }
        if !identity.ValidName(c.Local.ID) {
                return r, fmt.Errorf("%w: invalid local identity", ErrConfig)
        }
        if !identity.ValidUserID(string(c.Local.SM2UserID)) {
                return r, fmt.Errorf("%w: invalid local SM2 user identifier", ErrConfig)
        }
        if !identity.ValidName(c.Local.SignKeyID) || !identity.ValidName(c.Local.AgreeKeyID) {
                return r, fmt.Errorf("%w: invalid local key identifiers", ErrConfig)
        }
        if c.Local.SignKey == nil || c.Local.AgreeKey == nil {
                return r, fmt.Errorf("%w: local private keys are required", ErrConfig)
        }
        // A signing handle reused as an agreement handle would collapse two trust
        // roles into one. The two key types share a byte encoding, so the handle's
        // declared kind is what distinguishes them.
        sk, ok := c.Local.SignKey.(cryptosuite.KindedHandle)
        if !ok || sk.Kind() != cryptosuite.KeyKindSigning {
                return r, fmt.Errorf("%w: the signing slot does not hold a signing key", ErrConfig)
        }
        ak, ok := c.Local.AgreeKey.(cryptosuite.KindedHandle)
        if !ok || ak.Kind() != cryptosuite.KeyKindAgreement {
                return r, fmt.Errorf("%w: the agreement slot does not hold an agreement key", ErrConfig)
        }
        s, err := ValidateSuites(c.Suites)
        if err != nil {
                return r, err
        }
        r.suites = s
        for _, x := range s {
                r.u16 = append(r.u16, uint16(x))
        }
        b, err := record.PlaintextBudget(c.MaxFrame)
        if err != nil {
                return r, fmt.Errorf("%w: frame size leaves no room for a record", ErrConfig)
        }
        r.budget = b
        r.limit = c.RecordLimit
        if r.limit == 0 {
                r.limit = record.DefaultRecordLimit
        }
        if r.limit > record.DefaultRecordLimit {
                return r, fmt.Errorf("%w: record limit above the prototype bound", ErrConfig)
        }
        return r, nil
}

// lookup resolves an identity through the trusted registry and enforces the
// expected role and key identifiers. Anything unexpected is fatal.
func (r resolved) lookup(ctx context.Context, id, wantRole string) (identity.Peer, error) {
        if !identity.ValidName(id) {
                return identity.Peer{}, fmt.Errorf("%w: %q is not a legal identity", ErrPeerIdentity, id)
        }
        p, err := r.cfg.Trust.Lookup(ctx, id)
        if err != nil {
                return identity.Peer{}, fmt.Errorf("%w: %v", ErrPeerIdentity, err)
        }
        if p.Role != wantRole {
                return identity.Peer{}, fmt.Errorf("%w: %s is registered as %q, expected %q", ErrPeerIdentity, id, p.Role, wantRole)
        }
        return p, nil
}

func toSuites(u []uint16) []policy.Suite {
        out := make([]policy.Suite, 0, len(u))
        for _, v := range u {
                out = append(out, policy.Suite(v))
        }
        return out
}

// randomBytes reads from the cryptographic random source. A failure aborts the
// handshake; there is no fallback seed and no retry with weaker randomness.
func randomBytes(n int) ([]byte, error) {
        b := make([]byte, n)
        if _, err := rand.Read(b); err != nil {
                return nil, fmt.Errorf("handshake: local randomness unavailable: %w", err)
        }
        return b, nil
}

// frameIO owns framed message exchange during the handshake. It records the
// EXACT bytes written and read so the transcript authenticates what was really
// on the wire, not a re-encoding of parsed fields.
//
// DEADLINE (M2.1 fix): the deadline is computed ONCE for the whole handshake and
// is never extended. The M2 code reset a per-frame timeout on every message, so
// the caller's handshake deadline did not actually bound the network operations
// and a slow peer could keep the handshake alive indefinitely.
type frameIO struct {
        conn     net.Conn
        maxFrame uint32
        ctx      context.Context
        deadline time.Time // absolute, fixed for H1..H4
        gen      ioGuard
}

// newFrameIO fixes the handshake deadline: the earlier of the caller's deadline
// and the local budget. A parent context with no deadline still gets the local
// budget, because an unbounded handshake is never acceptable.
func newFrameIO(ctx context.Context, conn net.Conn, maxFrame uint32, budget time.Duration) (*frameIO, error) {
        d, err := handshakeDeadline(ctx, budget)
        if err != nil {
                return nil, err
        }
        return &frameIO{conn: conn, maxFrame: maxFrame, ctx: ctx, deadline: d}, nil
}

// arm installs the fixed deadline and a cancellation watcher for one message.
// The deadline is never moved forward, so H1..H4 share one budget.
func (f *frameIO) arm(write bool) (stop func()) {
        set := readDeadlineSetter(f.conn)
        if write {
                set = writeDeadlineSetter(f.conn)
        }
        return armIO(f.ctx, set, 0, f.deadline, &f.gen)
}

// fail classifies an I/O failure. A cancellation, a caller deadline and a local
// timeout are each reported as themselves; a malformed or missing message stays
// ErrProtocol; and a transport failure keeps its own cause.
func (f *frameIO) fail(what string, err error) error {
        if classified := operationError(f.ctx, err); classified != err {
                return classified
        }
        return fmt.Errorf("%w: %s: %v", ErrProtocol, what, err)
}

func (f *frameIO) send(kind wire.Kind, payload []byte) ([]byte, error) {
        enc, err := wire.Encode(wire.Frame{Kind: kind, Payload: payload}, f.maxFrame)
        if err != nil {
                return nil, fmt.Errorf("%w: encoding %s: %v", ErrProtocol, kindName(kind), err)
        }
        if err := cancelError(f.ctx); err != nil {
                return nil, err
        }
        stop := f.arm(true)
        defer stop()
        if err := wire.WriteEncoded(f.conn, enc); err != nil {
                return nil, f.fail("writing "+kindName(kind), err)
        }
        return enc, nil
}

func (f *frameIO) recv(want wire.Kind) (payload, raw []byte, err error) {
        if err := cancelError(f.ctx); err != nil {
                return nil, nil, err
        }
        stop := f.arm(false)
        defer stop()
        fr, raw, err := wire.ReadRaw(f.conn, f.maxFrame)
        if err != nil {
                return nil, nil, f.fail("reading "+kindName(want), err)
        }
        if fr.Kind != want {
                // This is where an out-of-order message, an early ProtectedRecord or an
                // injected lab message is caught: the kind must match exactly.
                return nil, nil, fmt.Errorf("%w: expected %s, received message kind %d", ErrProtocol, kindName(want), fr.Kind)
        }
        return fr.Payload, raw, nil
}

func kindName(k wire.Kind) string {
        switch k {
        case wire.ClientHello:
                return "ClientHello"
        case wire.ServerHello:
                return "ServerHello"
        case wire.ClientAuth:
                return "ClientAuth"
        case wire.ServerFinished:
                return "ServerFinished"
        case wire.ProtectedRecord:
                return "ProtectedRecord"
        case wire.LabTelemetry, wire.LabAck:
                return "lab-only message"
        }
        return "unknown message"
}

// Server runs the gateway side: SM2 responder, msg2 sender, msg4 sender.
//
// The returned Channel exists only after the device's signature AND the
// device's FinishedA have verified AND msg4 has been fully written. Writing
// msg4 successfully does NOT mean the device received or accepted it; that
// remains unknown here, and nothing in this package claims otherwise.
func Server(ctx context.Context, c Config) (Channel, error) {
        ch, err := serverHandshake(ctx, c)
        if err != nil {
                // c.Conn may be nil: configuration validation runs before any I/O, and a
                // caller that passed no connection still deserves a clean error rather
                // than a panic on the error path.
                if c.Conn != nil {
                        _ = c.Conn.Close()
                }
                return nil, err
        }
        return ch, nil
}

func serverHandshake(ctx context.Context, c Config) (Channel, error) {
        r, err := c.prepare(identity.RoleGateway)
        if err != nil {
                return nil, err
        }
        // ONE absolute deadline for the whole handshake, taken from the earlier of the
        // caller's deadline and the local budget. Messages never extend it.
        fio, err := newFrameIO(ctx, c.Conn, c.MaxFrame, c.Timeout)
        if err != nil {
                return nil, err
        }

        // --- msg1: ClientHello ---------------------------------------------------
        payload1, m1, err := fio.recv(wire.ClientHello)
        if err != nil {
                return nil, err
        }
        hello, err := parseClientHello(payload1)
        if err != nil {
                return nil, fmt.Errorf("%w: ClientHello: %v", ErrProtocol, err)
        }
        if hello.ResponderID != c.Local.ID {
                return nil, fmt.Errorf("%w: ClientHello is addressed to %q, not %q", ErrPeerIdentity, hello.ResponderID, c.Local.ID)
        }
        // The encapsulation key must match what this ClientHello ADVERTISES:
        // offering suite 2 requires a correctly sized key, offering only suite 1
        // requires the field to be empty. This is checked before the registry
        // lookup so a malformed advertisement cannot even reach that path.
        if err := checkClientKEMField(hello.OfferedSuites, hello.EphemeralKEM); err != nil {
                return nil, err
        }
        // The peer is resolved through the TRUSTED registry. A public key carried in
        // the message is never adopted; it must match what the registry already
        // holds for that identity, or the handshake stops.
        peer, err := r.lookup(ctx, hello.InitiatorID, identity.RoleDevice)
        if err != nil {
                return nil, err
        }
        if hello.InitiatorSignKeyID != peer.Public.SignKeyID || hello.InitiatorAgreeKeyID != peer.Public.AgreementKeyID {
                return nil, fmt.Errorf("%w: device presented key identifiers the registry does not hold", ErrPeerIdentity)
        }

        // Suite selection is provisional here and only becomes trustworthy once the
        // transcript is authenticated by the device's signature and FinishedA.
        sel, err := policy.Select(r.suites, toSuites(hello.OfferedSuites), c.MinimumMode, peer.MinimumMode, c.AllowLegacy)
        if err != nil {
                return nil, fmt.Errorf("%w: no mutually acceptable suite: %v", ErrUnsupported, err)
        }
        if err := requireSuite(uint16(sel)); err != nil {
                return nil, err
        }

        ag := cryptosuite.NewSM2Agreement()
        eph, err := ag.GenerateEphemeral()
        if err != nil {
                return nil, fmt.Errorf("%w: ephemeral SM2 key: %v", ErrProtocol, err)
        }
        defer func() { _ = eph.Private.Destroy() }()
        if len(eph.Public) != SM2PointSize {
                return nil, fmt.Errorf("%w: ephemeral SM2 key has length %d", ErrProtocol, len(eph.Public))
        }
        nb, err := randomBytes(NonceSize)
        if err != nil {
                return nil, err
        }

        // --- ML-KEM-768 encapsulation (suite 2 only) ----------------------------
        //
        // Real encapsulation against the key the device sent. The shared secret is
        // held for the key schedule; the ciphertext goes into ServerCore so that
        // SigB authenticates it and it becomes part of the transcript.
        var zpq, kemCT []byte
        if uint16(sel) == SuiteHybrid {
                zpq, kemCT, err = c.kem().Encapsulate(hello.EphemeralKEM)
                if err != nil {
                        return nil, fmt.Errorf("%w: ML-KEM encapsulation: %v", ErrProtocol, err)
                }
                if len(kemCT) != cryptosuite.MLKEM768CiphertextSize {
                        return nil, fmt.Errorf("%w: ML-KEM ciphertext has length %d",
                                ErrProtocol, len(kemCT))
                }
        }

        // --- SM2 key exchange and key schedule ----------------------------------
        peerID := cryptosuite.IdentityPublic{
                ID:              peer.Public.ID,
                SM2UserID:       peer.Public.SM2UserID,
                AgreementPublic: peer.Public.AgreementPublic,
        }
        localID := cryptosuite.IdentityPublic{ID: c.Local.ID, SM2UserID: c.Local.SM2UserID}
        if ph, ok := c.Local.AgreeKey.(cryptosuite.PublicHandle); ok {
                if pub, perr := ph.Public(); perr == nil {
                        localID.AgreementPublic = pub
                }
        }
        res, err := ag.Derive(cryptosuite.Responder, localID, c.Local.AgreeKey, eph, peerID, hello.EphemeralSM2)
        if err != nil {
                return nil, fmt.Errorf("%w: SM2 key exchange: %w", ErrProtocol, err)
        }
        defer clear(res.SharedSecret)
        defer clear(res.SendConfirmation)
        defer clear(res.ExpectedPeerConfirmation)

        // --- msg2: ServerCore + signature ---------------------------------------
        core := ServerCoreBody{
                SelectedSuite:       uint16(sel),
                SupportedSuites:     r.u16,
                ResponderID:         c.Local.ID,
                ResponderSignKeyID:  c.Local.SignKeyID,
                ResponderAgreeKeyID: c.Local.AgreeKeyID,
                InitiatorID:         hello.InitiatorID,
                Nonce:               nb,
                EphemeralSM2:        eph.Public,
                KEMCiphertext:       kemCT, // set only for suite 2
                SM2Confirmation:     res.SendConfirmation,
        }
        // Enforce the rule on what we are about to sign and send.
        if err := checkServerKEMField(uint16(sel), kemCT); err != nil {
                return nil, err
        }
        coreBytes := core.encode()
        sigB, err := cryptosuite.NewSM2Identity().Sign(
                c.Local.SignKey, c.Local.SM2UserID,
                serverAuthMessage(m1, coreBytes),
        )
        if err != nil {
                return nil, fmt.Errorf("%w: signing ServerHello: %v", ErrProtocol, err)
        }
        payload2 := make([]byte, 0, len(coreBytes)+len(sigB))
        payload2 = append(payload2, coreBytes...)
        payload2 = append(payload2, sigB...)
        m2, err := fio.send(wire.ServerHello, payload2)
        if err != nil {
                return nil, err
        }

        kikm, err := ikm(core.SelectedSuite, res.SharedSecret, zpq)
        if err != nil {
                return nil, err
        }
        slt, err := salt(hello.Nonce, nb)
        if err != nil {
                return nil, err
        }
        prk := cryptosuite.HKDFExtractSM3(slt, kikm)
        t2 := transcriptHashT2(m1, m2)
        kfinA, kfinB, err := finishedKeys(prk, t2)
        if err != nil {
                return nil, err
        }

        // --- msg3: verify the device --------------------------------------------
        payload3, m3, err := fio.recv(wire.ClientAuth)
        if err != nil {
                return nil, err
        }
        ccBytes, sigA, fa, err := splitClientAuth(payload3)
        if err != nil {
                return nil, fmt.Errorf("%w: ClientAuth is too short", ErrProtocol)
        }
        cc, err := parseClientCore(ccBytes)
        if err != nil {
                return nil, fmt.Errorf("%w: ClientCore: %v", ErrProtocol, err)
        }
        if cc.InitiatorID != hello.InitiatorID || cc.InitiatorID != peer.Public.ID {
                return nil, fmt.Errorf("%w: ClientAuth names a different device", ErrPeerIdentity)
        }
        if cc.InitiatorSignKeyID != peer.Public.SignKeyID || cc.InitiatorAgreeKeyID != peer.Public.AgreementKeyID {
                return nil, fmt.Errorf("%w: ClientAuth presents unexpected key identifiers", ErrPeerIdentity)
        }
        if cc.SelectedSuite != core.SelectedSuite {
                return nil, fmt.Errorf("%w: device acknowledged suite %d, expected %d", ErrProtocol, cc.SelectedSuite, core.SelectedSuite)
        }
        if err := cryptosuite.NewSM2Identity().Verify(
                peer.Public.SignPublic, peer.Public.SM2UserID,
                clientAuthMessage(m1, m2, ccBytes), sigA,
        ); err != nil {
                return nil, fmt.Errorf("%w: device signature: %v", ErrAuthentication, err)
        }
        if len(res.ExpectedPeerConfirmation) != cryptosuite.SM3Size || !cryptosuite.ConstantTimeEqual(cc.SM2Confirmation, res.ExpectedPeerConfirmation) {
                return nil, fmt.Errorf("%w: device S2", ErrSM2Confirmation)
        }
        wantA := confirmA(kfinA, m1, m2, ccBytes, sigA)
        if !cryptosuite.ConstantTimeEqual(fa, wantA) {
                return nil, fmt.Errorf("%w: device Finished did not match", ErrTranscript)
        }

        // --- msg4: ServerFinished ------------------------------------------------
        fb := confirmB(kfinB, m1, m2, m3)
        m4, err := fio.send(wire.ServerFinished, fb)
        if err != nil {
                return nil, err
        }

        // msg4 has been written. Only now does the gateway hold a usable session.
        t4 := transcriptHashT4(m1, m2, m3, m4)
        sm, err := sessionKeys(prk, t4)
        if err != nil {
                return nil, err
        }
        // Record what THIS handshake actually verified. The registry may be rotated
        // a moment later; the service compares this snapshot with the latest entry
        // before admitting business traffic, so a session authenticated on
        // generation N is never registered as N+1.
        snap := PeerSnapshot{
                PeerID:           peer.Public.ID,
                SignKeyID:        peer.Public.SignKeyID,
                AgreementKeyID:   peer.Public.AgreementKeyID,
                KeyGeneration:    peer.KeyGeneration,
                RegistryRevision: peer.Revision,
        }
        ch, err := newChannel(c.Conn, c.MaxFrame, c.Timeout, sm, core.SelectedSuite, peer.Public.ID, snap, false, r.limit, r.budget)
        return profileChannel(ch, err, c.IdleTimeout)
}

// This file holds the cancellation and deadline primitives.
//
// The M2 review found that cancellation never reached the actual I/O:
// Send/Receive only checked ctx once, before taking a sync.Mutex, and the
// handshake reset its timeout on every frame. A blocked Read, a blocked Write,
// a goroutine queueing for the send lock and a stalled handshake were all
// uninterruptible, and the outer handshake deadline did not bound the network
// operations at all.
//
// The rules implemented here:
//
//  1. Waiting for serialisation permission is cancellable, and a waiter that is
//     cancelled before it starts must NOT go on to send.
//  2. An operation that has already started is interrupted by cancellation via
//     an expired deadline, because a net.Conn offers no other portable way to
//     unblock a Read or a Write.
//  3. Only the caller's context ending produces a context error. A peer close, a
//     truncated frame or a transport failure keeps its own cause.
//  4. A stale cancellation callback must never expire the deadline of a LATER
//     operation. context.AfterFunc's stop does not wait for a callback that has
//     already begun, so each operation carries a generation number and a
//     callback only acts while its generation is current.

// --- cancellable serialisation ---------------------------------------------

// semaphore is a one-slot mutex that can be waited for under a context.
//
// sync.Mutex cannot do this: a goroutine blocked in Lock() is uninterruptible,
// so a cancelled Send or Receive could sit in the queue for as long as the
// current holder takes. The project needs "waiting for permission is itself
// cancellable", which sync.Mutex cannot express.
type semaphore chan struct{}

func newSemaphore() semaphore { return make(semaphore, 1) }

// acquire takes the slot or returns ctx.Err(). On error the slot was NOT taken,
// so the caller must not release it.
func (s semaphore) acquire(ctx context.Context) error {
        if err := ctx.Err(); err != nil {
                return err
        }
        select {
        case s <- struct{}{}:
                return nil
        case <-ctx.Done():
                return ctx.Err()
        }
}

func (s semaphore) release() { <-s }

// --- per-operation deadlines -----------------------------------------------

// deadlineSetter writes either the read or the write deadline. Read and write
// deadlines are kept separate on purpose: a send must never expire the deadline
// of a receive that is in flight at the same time on the same connection.
type deadlineSetter func(time.Time)

func readDeadlineSetter(conn net.Conn) deadlineSetter {
        return func(t time.Time) { _ = conn.SetReadDeadline(t) }
}

func writeDeadlineSetter(conn net.Conn) deadlineSetter {
        return func(t time.Time) { _ = conn.SetWriteDeadline(t) }
}

// ioDeadline returns the deadline for one operation: the earlier of the caller's
// context deadline and the configured timeout. A caller asking for less time
// than the local default gets less time.
func ioDeadline(ctx context.Context, timeout time.Duration) time.Time {
        d := time.Now().Add(timeout)
        if cd, ok := ctx.Deadline(); ok && cd.Before(d) {
                return cd
        }
        return d
}

// armIO installs a deadline and arranges for context cancellation to interrupt
// an operation already in flight. The returned stop function must be called when
// the operation ends.
//
// fixed may be non-zero to pin the deadline to one absolute instant; that is how
// the handshake enforces a single budget across all of H1..H4.
// ioGuard serialises deadline arming with cancellation callbacks.
//
// R2 (集中自查): a bare generation counter could not close the interleaving the
// review described. context.AfterFunc's Stop does NOT wait for a callback that
// is already running, so a callback that had passed its generation check and was
// then descheduled could still apply "deadline = now" AFTER the next operation
// had installed its own deadline -- silently shortening it. A deterministic test
// (timeout_deadline_interleave_test.go) reproduced exactly that before this fix.
//
// The guard holds a mutex that covers three critical sections: installing an
// operation's deadline, a cancellation callback applying "now", and ending an
// operation (which invalidates its callback). With every path under the same
// mutex, a stale callback can never interleave with the next operation's
// installation: it either completes while its operation is still current, or it
// sees a bumped generation and does nothing.
//
// Deadlock safety: the mutex is only ever held across conn.Set{Read,Write}
// Deadline calls, which return immediately even on a closed or blocked
// connection, and stop() taking the mutex can therefore only wait for a
// bounded, fast setter -- never for network I/O.
type ioGuard struct {
        mu  sync.Mutex
        seq uint64
}

func (g *ioGuard) claimLocked() uint64 {
        g.seq++
        return g.seq
}

func armIO(ctx context.Context, set deadlineSetter, timeout time.Duration, fixed time.Time, g *ioGuard) (stop func()) {
        g.mu.Lock()
        mine := g.claimLocked()
        if fixed.IsZero() {
                set(ioDeadline(ctx, timeout))
        } else {
                set(fixed)
        }
        stopFn := context.AfterFunc(ctx, func() {
                g.mu.Lock()
                defer g.mu.Unlock()
                if g.seq == mine {
                        // Expire the deadline now so a blocked Read or Write returns at
                        // once instead of waiting for the network or the timeout.
                        set(time.Now())
                }
        })
        g.mu.Unlock()
        return func() {
                // Invalidate first: any callback that has not yet taken the mutex sees a
                // bumped generation and does nothing. A callback that is already inside
                // its critical section finishes first (its set is bounded and fast), and
                // the next operation can only install its own deadline afterwards.
                g.mu.Lock()
                g.seq++
                g.mu.Unlock()
                stopFn()
        }
}

// --- error classification ---------------------------------------------------

// cancelError returns ctx.Err() when the caller's context has ended, and nil
// otherwise.
//
// This is the ONLY place a context error is produced. Everything else keeps its
// own cause, so callers can still distinguish "I cancelled this" from "the
// network broke" or "the peer closed".
func cancelError(ctx context.Context) error {
        if ctx == nil {
                return nil
        }
        return ctx.Err()
}

// operationError classifies the failure of an I/O operation that was underway.
//
// Precedence:
//  1. the caller's context ended                     -> ctx.Err()
//     (context.Canceled or context.DeadlineExceeded)
//  2. a deadline has passed                          -> context.DeadlineExceeded
//     when it was the CALLER's deadline, otherwise ErrTimeout
//  3. anything else                                  -> the original error,
//     unchanged, so a peer close or a transport failure keeps its own cause
//
// Case 2 covers a real race: the transport can report a timeout a few
// microseconds before the context package publishes Err(). Without the explicit
// deadline check the SAME condition would be reported as a protocol error or as
// a deadline depending on scheduling.
func operationError(ctx context.Context, err error) error {
        if err == nil {
                return nil
        }
        if cerr := cancelError(ctx); cerr != nil {
                return cerr
        }
        if isTimeout(err) {
                if cd, ok := ctx.Deadline(); ok && !time.Now().Before(cd) {
                        return context.DeadlineExceeded
                }
                return fmt.Errorf("%w: %v", ErrTimeout, err)
        }
        return err
}

// isTimeout reports whether err came from an expired deadline rather than from
// the peer.
func isTimeout(err error) bool {
        var ne net.Error
        if errors.As(err, &ne) && ne.Timeout() {
                return true
        }
        return errors.Is(err, os.ErrDeadlineExceeded)
}

// --- handshake budget -------------------------------------------------------

// handshakeDeadline computes the SINGLE absolute deadline for an entire
// handshake: the earlier of the caller's deadline and the local handshake
// budget. If the caller supplied no deadline, the local budget still applies —
// an unbounded handshake is never acceptable.
//
// Later messages must NOT extend it. A peer that dribbles out one byte per
// interval, or that stalls between H2 and H3, must not be able to buy the
// handshake more time than the budget allows.
func handshakeDeadline(ctx context.Context, budget time.Duration) (time.Time, error) {
        if budget <= 0 {
                return time.Time{}, fmt.Errorf("%w: handshake budget must be positive", ErrConfig)
        }
        now := time.Now()
        d := now.Add(budget)
        if cd, ok := ctx.Deadline(); ok && cd.Before(d) {
                d = cd
        }
        if !d.After(now) {
                return time.Time{}, context.DeadlineExceeded
        }
        return d, nil
}

// Package policy handles local policy only. Selection is provisional until the
// complete handshake transcript is authenticated by BOTH peers.
package policy

import "errors"

type Mode string

const (
        Traditional Mode = "traditional"
        Hybrid      Mode = "hybrid"
)

type Suite uint16

const (
        TraditionalSM    Suite = 1
        HybridSMMLKEM768 Suite = 2
)

var ErrPolicy = errors.New("mode or suite rejected by enrolled policy")

func valid(m Mode) bool { return m == Traditional || m == Hybrid }
func check(s []Suite) bool {
        if len(s) == 0 || len(s) > 2 {
                return false
        }
        seen := map[Suite]bool{}
        for _, v := range s {
                if (v != TraditionalSM && v != HybridSMMLKEM768) || seen[v] {
                        return false
                }
                seen[v] = true
        }
        return true
}
func has(s []Suite, v Suite) bool {
        for _, x := range s {
                if x == v {
                        return true
                }
        }
        return false
}

// ModeOf reports which policy tier a negotiated suite belongs to.
//
// Admission uses this to re-check that a session still satisfies the CURRENT
// trusted minimum mode. The suite is fixed at handshake time, but the policy is
// not: an operator can raise a device's minimum from Traditional to Hybrid while
// a Suite 1 handshake is already in flight, and such a session must not be
// admitted on keys it proved under the old policy.
func ModeOf(s Suite) (Mode, error) {
        switch s {
        case TraditionalSM:
                return Traditional, nil
        case HybridSMMLKEM768:
                return Hybrid, nil
        default:
                return "", ErrPolicy // unknown suite
        }
}

// Satisfies reports whether a negotiated suite meets a required minimum mode.
func Satisfies(s Suite, minimum Mode) bool {
        got, err := ModeOf(s)
        if err != nil {
                return false
        }
        return got == minimum || (got == Hybrid && minimum == Traditional)
}

// peerMinimum MUST come from a trusted registry, never from unauthenticated input.
func Select(local, peer []Suite, localMinimum, peerMinimum Mode, allowLegacy bool) (Suite, error) {
        if !valid(localMinimum) || !valid(peerMinimum) || !check(local) || !check(peer) {
                return 0, ErrPolicy // unknown suite
        }
        if has(local, HybridSMMLKEM768) && has(peer, HybridSMMLKEM768) {
                return HybridSMMLKEM768, nil
        }
        if localMinimum == Hybrid || peerMinimum == Hybrid || !allowLegacy {
                return 0, ErrPolicy // unknown suite
        }
        if has(local, TraditionalSM) && has(peer, TraditionalSM) {
                return TraditionalSM, nil
        }
        return 0, ErrPolicy
}

// Package record supplies a low-level record framing experiment with an injected
// AEAD. It does NOT perform authentication, select SM4, or derive session keys.
package record

import (
        "crypto/cipher"
        "encoding/binary"
        "errors"
        "sync"
)

const HeaderSize = 33
const MaxPlaintext = 16 * 1024
const DefaultRecordLimit uint64 = 1 << 20 // prototype policy, not a universal GCM security bound

// TagSize is the AEAD tag length this record format requires (SM4-GCM, 16 bytes).
const TagSize = 16

// FrameOverhead is the number of bytes a protected record adds on top of the
// application plaintext when it sits inside a wire frame:
//
//        wire frame body = version(1) + kind(1) + payload
//        payload         = record header(33) + ciphertext(len(pt)) + tag(16)
//
// so plaintext = frame body - 2 - 33 - 16. The outer 4-byte length prefix that
// wire.Encode writes is NOT part of max_frame_bytes and is therefore not
// subtracted here.
const FrameOverhead = 2 + HeaderSize + TagSize

// PlaintextBudget returns the largest application plaintext that still fits in
// one protected record inside a frame body limited to maxFrameBytes bytes. It
// is the single place where the frame limit and the record limit are reconciled,
// so the two can never drift apart.
func PlaintextBudget(maxFrameBytes uint32) (int, error) {
        if maxFrameBytes <= FrameOverhead {
                return 0, ErrRecord
        }
        b := int(maxFrameBytes) - FrameOverhead
        if b > MaxPlaintext {
                b = MaxPlaintext
        }
        if b < 1 {
                return 0, ErrRecord
        }
        return b, nil
}

const (
        Telemetry byte = 1
        Control   byte = 2
        Ack       byte = 3
)
const (
        DeviceToGateway byte = 1
        GatewayToDevice byte = 2
)

var (
        ErrRecord         = errors.New("invalid record")
        ErrReplayOrOrder  = errors.New("repeated or out-of-order sequence")
        ErrLimit          = errors.New("record limit reached; fresh full handshake required")
        ErrAuthentication = errors.New("record authentication failed")
)

type Meta struct {
        Suite     uint16
        SessionID [16]byte
        Epoch     uint32
        Direction byte
}

func (m Meta) valid() bool {
        return (m.Suite == 1 || m.Suite == 2) && (m.Direction == DeviceToGateway || m.Direction == GatewayToDevice) && m.SessionID != [16]byte{}
}
func header(m Meta, kind byte, seq uint64) []byte {
        b := make([]byte, HeaderSize)
        b[0] = 1
        binary.BigEndian.PutUint16(b[1:3], m.Suite)
        b[3] = kind
        b[4] = m.Direction
        copy(b[5:21], m.SessionID[:])
        binary.BigEndian.PutUint32(b[21:25], m.Epoch)
        binary.BigEndian.PutUint64(b[25:33], seq)
        return b
}
func nonce(prefix [4]byte, seq uint64) []byte {
        b := make([]byte, 12)
        copy(b, prefix[:])
        binary.BigEndian.PutUint64(b[4:], seq)
        return b
}
func goodType(k byte) bool { return k == Telemetry || k == Control || k == Ack }
func valid(a cipher.AEAD, m Meta, limit uint64, budget int) bool {
        return a != nil && a.NonceSize() == 12 && a.Overhead() == TagSize && m.valid() && limit > 0 && limit <= DefaultRecordLimit && budget >= 1 && budget <= MaxPlaintext
}

// Use distinct directional AEAD keys. Never reconstruct a Sender/Receiver with
// old keys and reset sequence numbers. Production network writes need one lock
// around BOTH Seal and wire.Write to preserve TCP order.
type Sender struct {
        mu          sync.Mutex
        a           cipher.AEAD
        m           Meta
        prefix      [4]byte
        next, limit uint64
        budget      int
}
type Receiver struct {
        mu          sync.Mutex
        a           cipher.AEAD
        m           Meta
        prefix      [4]byte
        next, limit uint64
        budget      int
}

func NewSender(a cipher.AEAD, m Meta, prefix [4]byte, limit uint64) (*Sender, error) {
        return NewSenderBudget(a, m, prefix, limit, MaxPlaintext)
}
func NewReceiver(a cipher.AEAD, m Meta, prefix [4]byte, limit uint64) (*Receiver, error) {
        return NewReceiverBudget(a, m, prefix, limit, MaxPlaintext)
}

// NewSenderBudget builds a Sender whose plaintext limit is derived from the
// configured frame size (see PlaintextBudget) rather than assumed.
func NewSenderBudget(a cipher.AEAD, m Meta, prefix [4]byte, limit uint64, budget int) (*Sender, error) {
        if !valid(a, m, limit, budget) {
                return nil, ErrRecord
        }
        return &Sender{a: a, m: m, prefix: prefix, limit: limit, budget: budget}, nil
}

// NewReceiverBudget is NewSenderBudget for the receiving direction.
func NewReceiverBudget(a cipher.AEAD, m Meta, prefix [4]byte, limit uint64, budget int) (*Receiver, error) {
        if !valid(a, m, limit, budget) {
                return nil, ErrRecord
        }
        return &Receiver{a: a, m: m, prefix: prefix, limit: limit, budget: budget}, nil
}
func (s *Sender) Seal(kind byte, plaintext []byte) ([]byte, error) {
        s.mu.Lock()
        defer s.mu.Unlock()
        if !goodType(kind) || len(plaintext) > s.budget {
                return nil, ErrRecord
        }
        if s.next >= s.limit {
                return nil, ErrLimit
        }
        h := header(s.m, kind, s.next)
        out := s.a.Seal(nil, nonce(s.prefix, s.next), plaintext, h)
        s.next++ // consumed even if the later network write fails; close that session
        return append(h, out...), nil
}
func (r *Receiver) Open(b []byte) (byte, []byte, error) {
        r.mu.Lock()
        defer r.mu.Unlock()
        if len(b) < HeaderSize+r.a.Overhead() || len(b) > HeaderSize+r.budget+r.a.Overhead() {
                return 0, nil, ErrRecord
        }
        kind := b[3]
        seq := binary.BigEndian.Uint64(b[25:33])
        want := header(r.m, kind, seq)
        if !goodType(kind) {
                return 0, nil, ErrRecord
        }
        for i := 0; i < HeaderSize; i++ {
                if b[i] != want[i] {
                        return 0, nil, ErrRecord
                }
        }
        if r.next >= r.limit {
                return 0, nil, ErrLimit
        }
        if seq != r.next {
                return 0, nil, ErrReplayOrOrder
        }
        p, e := r.a.Open(nil, nonce(r.prefix, seq), b[HeaderSize:], b[:HeaderSize])
        if e != nil {
                return 0, nil, ErrAuthentication
        }
        r.next++ // authenticated acceptance is the ONLY place receive state advances
        return kind, p, nil
}

// Package wire implements bounded framing, NOT a secure channel.
package wire

import (
        "encoding/binary"
        "errors"
        "fmt"
        "io"
)

const Version byte = 1
const HardMaxFrame uint32 = 64 * 1024

type Kind byte

const (
        ClientHello     Kind = 1
        ServerHello     Kind = 2
        ClientAuth      Kind = 3
        ServerFinished  Kind = 4
        ProtectedRecord Kind = 16
        LabTelemetry    Kind = 240
        LabAck          Kind = 241
)

var ErrFrame = errors.New("invalid or oversized frame")

type Frame struct {
        Kind    Kind
        Payload []byte
}

func validKind(k Kind) bool {
        switch k {
        case ClientHello, ServerHello, ClientAuth, ServerFinished, ProtectedRecord, LabTelemetry, LabAck:
                return true
        }
        return false
}

// Encode: uint32(big endian, remaining length) || version || kind || payload.
// Handshakes must authenticate these exact encoded bytes, not re-marshaled objects.
func Encode(f Frame, max uint32) ([]byte, error) {
        if max < 2 || max > HardMaxFrame || !validKind(f.Kind) || len(f.Payload) > int(max)-2 {
                return nil, ErrFrame
        }
        out := make([]byte, 6+len(f.Payload))
        binary.BigEndian.PutUint32(out[:4], uint32(len(f.Payload)+2))
        out[4] = Version
        out[5] = byte(f.Kind)
        copy(out[6:], f.Payload)
        return out, nil
}

// WriteEncoded writes an already-encoded frame, looping until every byte is
// delivered. Use this together with Encode when the exact bytes put on the wire
// must also be captured (for example, as a handshake transcript input): encode
// once, hand the same slice to the transcript and to this writer.
func WriteEncoded(w io.Writer, b []byte) error {
        for len(b) > 0 {
                n, e := w.Write(b)
                if n < 0 || n > len(b) {
                        return io.ErrShortWrite
                }
                b = b[n:]
                if e != nil {
                        return e
                }
                if n == 0 {
                        return io.ErrShortWrite
                }
        }
        return nil
}

func Write(w io.Writer, f Frame, max uint32) error {
        b, err := Encode(f, max)
        if err != nil {
                return err
        }
        return WriteEncoded(w, b)
}

// ErrTruncated means a frame had already started and the stream ended before it
// was complete, including a partial length prefix. This is NOT a clean end of stream and
// must never be reported as one — a peer that stops mid-frame may have been
// killed, or a record may have been cut short by an attacker.
//
// It deliberately wraps io.ErrUnexpectedEOF so callers can test either way:
// errors.Is(err, wire.ErrTruncated) for the project-specific classification, or
// errors.Is(err, io.ErrUnexpectedEOF) for the standard idiom.
var ErrTruncated = fmt.Errorf("frame truncated: stream ended mid-frame: %w", io.ErrUnexpectedEOF)

// ReadRaw behaves like Read but also returns the exact bytes that were read off
// the wire, including the outer length prefix. A receiver must authenticate
// what it actually received rather than a re-encoding of the parsed fields.
//
// EOF CLASSIFICATION (M2.1 fix): the three ways a stream can end are now
// distinguished, because conflating them hid a real defect:
//
//   - nothing read at all            -> io.EOF          (clean end between frames)
//   - part of the length prefix read -> ErrTruncated    (frame had started)
//   - prefix read, body missing/part -> ErrTruncated    (frame had started)
//
// Before the fix the last case returned io.EOF, so a peer that died mid-frame
// looked identical to a peer that finished normally.
func ReadRaw(r io.Reader, max uint32) (Frame, []byte, error) {
        if max < 2 || max > HardMaxFrame {
                return Frame{}, nil, ErrFrame
        }
        var prefix [4]byte
        if _, err := io.ReadFull(r, prefix[:]); err != nil {
                if errors.Is(err, io.ErrUnexpectedEOF) {
                        return Frame{}, nil, ErrTruncated
                }
                return Frame{}, nil, err
        }
        n := binary.BigEndian.Uint32(prefix[:])
        if n < 2 || n > max {
                return Frame{}, nil, ErrFrame
        }
        body := make([]byte, int(n))
        if _, err := io.ReadFull(r, body); err != nil {
                // The prefix was FULLY read: a frame has definitely started, so any
                // shortfall here is a truncation, including the case where io.ReadFull
                // reports a plain io.EOF because zero body bytes arrived.
                if errors.Is(err, io.EOF) || errors.Is(err, io.ErrUnexpectedEOF) {
                        return Frame{}, nil, ErrTruncated
                }
                return Frame{}, nil, err
        }
        if body[0] != Version || !validKind(Kind(body[1])) {
                return Frame{}, nil, ErrFrame
        }
        raw := make([]byte, 0, 4+len(body))
        raw = append(raw, prefix[:]...)
        raw = append(raw, body...)
        return Frame{Kind: Kind(body[1]), Payload: body[2:]}, raw, nil
}

func Read(r io.Reader, max uint32) (Frame, error) {
        f, _, err := ReadRaw(r, max)
        return f, err
}
